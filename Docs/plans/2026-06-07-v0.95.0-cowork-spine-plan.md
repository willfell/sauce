```js
/**
 * composeFinalPreferences({ bundle, overrides, ad_hoc_prefs })
 *
 * Composes the FINAL preferences tree from layered inputs:
 *   bundle:        engagement-type JSON parsed (e.g. personal.json) — required
 *   overrides:     engagement.overrides block from vault-config.md (optional; may be absent or empty)
 *   ad_hoc_prefs:  optional runtime overrides (default null; reserved for future use)
 *
 * Composition order:
 *   - Objects (render_aspects, voice, microscopes_registry): bundle ⨁ overrides → MERGE
 *   - Arrays (cadence_order entries, tripwire_aspects): overrides REPLACES bundle when present
 *
 * Returns:
 *   {
 *     render_aspects:       {...},       // merged
 *     cadence_order:        {...},       // per-cadence; bundle provides defaults; overrides replaces per-cadence
 *     voice:                {...},       // merged
 *     microscopes_registry: {...} | null, // null when neither bundle nor overrides specify; triggers filesystem scan
 *     tripwire_aspects:     [...],       // overrides REPLACES bundle's array
 *   }
 *
 * Backward-compat: when overrides is absent/empty/malformed, returns bundle defaults verbatim.
 */
function composeFinalPreferences({ bundle, overrides, ad_hoc_prefs } = {}) {
    if (!bundle || typeof bundle !== "object") {
        return { render_aspects: {}, cadence_order: {}, voice: {}, microscopes_registry: null, tripwire_aspects: [] };
    }
    // Defensive: overrides must be a plain object; otherwise treat as absent + warn (caller emits Notice).
    const ov = (overrides && typeof overrides === "object" && !Array.isArray(overrides)) ? overrides : {};

    const final = {
        render_aspects:   Object.assign({}, bundle.render_aspects || {}, ov.render_aspects || {}),
        cadence_order:    Object.assign({}, bundle.cadence_order   || {}, ov.cadence_order   || {}),
        voice:            Object.assign({}, bundle.voice           || {}, ov.voice           || {}),
        // microscopes_registry: bundle null + overrides null => null (filesystem scan fallback)
        microscopes_registry: ov.microscopes_registry || bundle.microscopes_registry || null,
        // tripwire_aspects: array — overrides REPLACES (not merges)
        tripwire_aspects: Array.isArray(ov.tripwire_aspects)
            ? ov.tripwire_aspects.slice()
            : (Array.isArray(bundle.tripwire_aspects) ? bundle.tripwire_aspects.slice() : []),
    };
    return final;
}
```

```js
/**
 * loadKindTitles({ vault_root })
 *
 * Loads the canonical kind→title map from spice/cowork/data/kind-titles.json.
 * Falls back to module-private CANONICAL_TITLES const when the file is missing
 * or older than v1.0.0.
 *
 * Returns: { kind_name: title, ... }
 */
function loadKindTitles({ vault_root } = {}) {
    if (!vault_root) return Object.assign({}, CANONICAL_TITLES);
    const path = require("path");
    const fs   = require("fs");
    const dataPath = path.join(vault_root, "spice/cowork/data/kind-titles.json");
    try {
        if (!fs.existsSync(dataPath)) return Object.assign({}, CANONICAL_TITLES);
        const data = JSON.parse(fs.readFileSync(dataPath, "utf8"));
        if (data.version !== "1.0.0" || !data.canonical_titles) return Object.assign({}, CANONICAL_TITLES);
        return Object.assign({}, data.canonical_titles);
    } catch (e) {
        return Object.assign({}, CANONICAL_TITLES);
    }
}
```
