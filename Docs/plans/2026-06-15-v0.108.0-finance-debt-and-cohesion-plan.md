async function applyFinanceBudgetGroupSeed(tp, manifest, variables, history, git) {
  if (!manifest || manifest.name !== "finance") return;
  if (!tp || !tp.app || !tp.app.vault || !tp.app.vault.adapter) return;
  const adapter = tp.app.vault.adapter;

  const budgetsRoot = "spice/finance/budgets";
  if (!(await adapter.exists(budgetsRoot))) return;

  // Read Budget Defaults
  const defaultsPath = "spice/finance/Budget Defaults.md";
  if (!(await adapter.exists(defaultsPath))) return;
  const defaultsBody = await adapter.read(defaultsPath);
  const defaults = _parseFrontmatterStrict(defaultsBody);
  if (!defaults || !Array.isArray(defaults.groups) || !Array.isArray(defaults.categories)) return;
  if (defaults.groups.length === 0) return;  // nothing to seed from

  // Walk all Budget-*.md
  const budgetFiles = await _listBudgetFiles(adapter, budgetsRoot);
  if (budgetFiles.length === 0) return;

  // Build category→group map from defaults
  const catToGroup = new Map();
  for (const c of defaults.categories) {
    if (c && c.name && c.group) catToGroup.set(c.name.toLowerCase(), c.group);
  }

  // Snapshot before mutation
  const snapshotRoot = await _ensureSauceBackup(adapter);  // existing helper from CF-1
  let groupsSeeded = 0;
  let categoriesReassigned = 0;

  for (const fp of budgetFiles) {
    try {
      const body = await adapter.read(fp);
      if (/__group_seed_migrated:\s*v0\.108\.0/.test(body)) continue;  // idempotency
      const { body: out, touched, seededGroups, reassignedCats } =
        _seedBudgetGroups(body, defaults.groups, catToGroup);
      if (!touched) continue;
      await _snapshotFile(adapter, snapshotRoot, fp);

      groupsSeeded += seededGroups;
      categoriesReassigned += reassignedCats;
      history?.push({ event: "info", step: "finance_budget_group_seed", name: "finance",

        attempted_at: new Date().toISOString() });

async function applyFinancePaycheckDefaultsDebtLinking(tp, manifest, variables, history, git) {
  if (!manifest || manifest.name !== "finance") return;
  if (!tp || !tp.app || !tp.app.vault || !tp.app.vault.adapter) return;
  const adapter = tp.app.vault.adapter;

  const paycheckDefaultsPath = "spice/finance/Paycheck Defaults.md";

  const body = await adapter.read(paycheckDefaultsPath);
  if (/__debt_links_migrated:\s*v0\.108\.0/.test(body)) return;  // idempotency

  // Walk debt entities to build name→file map
  const debtsRoot = "spice/finance/debts";
  if (!(await adapter.exists(debtsRoot))) return;
  const debtListing = await adapter.list(debtsRoot);
  const debtFiles = (debtListing.files || []).filter(p => /Debt-[^/]+\.md$/.test(p));
  const debtNameToSlug = new Map();
  for (const dp of debtFiles) {
    const debtBody = await adapter.read(dp);
    const fm = _parseFrontmatterStrict(debtBody);
    if (!fm || !fm.name) continue;
    const slug = path.basename(dp, ".md");  // e.g. "Debt-Apple-Card"
    debtNameToSlug.set(fm.name.toLowerCase(), slug);
  }
  if (debtNameToSlug.size === 0) return;  // no debts yet — nothing to link

  // Snapshot before mutation
  const snapshotRoot = await _ensureSauceBackup(adapter);
  await _snapshotFile(adapter, snapshotRoot, paycheckDefaultsPath);

  await adapter.write(paycheckDefaultsPath, result.body);

```javascript
async _resolveDebt(linkStr) {
  // linkStr e.g. "[[Debt-Apple-Card]]"
  const m = typeof linkStr === "string" ? linkStr.match(/^\[\[(.+?)\]\]$/) : null;
  if (!m) return null;
  const name = m[1];
  const dest = app.metadataCache.getFirstLinkpathDest(name, "");
  if (!dest) return null;
  const cache = app.metadataCache.getFileCache(dest);
  return cache?.frontmatter || null;
}
```

- [ ] **S4.3** — Extend the row render loop. After the Category cell, if `row.debt`, render a chip:

if (row.debt) {
  const debt = await this._resolveDebt(row.debt);
  if (debt) {
    const progressPct = (debt.kind === "credit-card" && debt.credit_limit > 0)
      ? Math.round((1 - debt.current_balance / debt.credit_limit) * 100)
      : null;
    const pct = progressPct !== null ? `${progressPct}% paid` : `debt`;
    const color = progressPct === null ? "#6b7280"  // grey for non-cc
      : progressPct < 33 ? "#dc2626" : progressPct < 66 ? "#b45309" : "#16a34a";
    const chip = cell.createEl("span", {

class DebtSummary {
  async render(dv) {
    if (dv.container.closest(".markdown-embed")) {
      const prev = dv.container.querySelector(".dbt-sum-root");
      if (prev) prev.remove();
    }

    const page = dv.current() || {};
    if (page.type !== "debt") return;

    const root = dv.container.createEl("div", { cls: "dbt-sum-root" });

    const balance = Number(page.current_balance) || 0;
    const apr = Number(page.apr) || 0;
    const planned = Number(page.planned_monthly_payment) || 0;
    const monthlyInterest = balance * apr / 100 / 12;
    const principalAttack = planned - monthlyInterest;

    // Band 1 — numbers
    const b1 = root.createEl("div", { cls: "dbt-band-1" });
    b1.createEl("div", { cls: "dbt-stat", text: `Balance: $${balance.toFixed(2)}` });
    b1.createEl("div", { cls: "dbt-stat", text: `APR: ${apr.toFixed(2)}%` });
    b1.createEl("div", { cls: "dbt-stat", text: `Monthly interest: $${monthlyInterest.toFixed(2)}` });
    b1.createEl("div", { cls: "dbt-stat", text: `Planned attack: $${planned.toFixed(2)}` });

    if (principalAttack <= 0) {
      b1.createEl("div", { cls: "dbt-warn", text: "Increase planned monthly attack — below interest" })
        .style.color = "#dc2626";
    } else {
      const months = Math.ceil(balance / principalAttack);
      const eta = new Date();
      eta.setMonth(eta.getMonth() + months);
      b1.createEl("div", { cls: "dbt-stat", text: `Projected payoff: ${months} months (${eta.toISOString().slice(0, 10)})` });
    }

    // Edit balance pill (triggers DebtConfigEditor)
    const editBtn = b1.createEl("button", { cls: "dbt-edit", text: "Edit balance" });
    editBtn.addEventListener("click", async () => {
      const file = app.vault.getAbstractFileByPath(page.file.path);
      await customJS.DebtConfigEditor.render(file, { onSave: () => { /* re-render */ } });
    });

    // Band 2 — sparkline + delta pill
    const history = Array.isArray(page.balance_history) ? page.balance_history : [];
    const b2 = root.createEl("div", { cls: "dbt-band-2" });
    if (history.length < 2) {
      b2.createEl("div", { cls: "dbt-placeholder", text: "No history yet — edit balance to start tracking" });
    } else {
      // Render inline SVG sparkline from history[].balance
      this._renderSparkline(b2, history.map(h => Number(h.balance)).reverse());
      const delta = balance - Number(history[0].balance);
      const deltaPill = b2.createEl("span", { cls: "dbt-delta", text: `Δ ${delta >= 0 ? "+" : ""}$${delta.toFixed(2)}` });
      deltaPill.style.color = delta > 0 ? "#dc2626" : "#16a34a";
    }

    // Band 3 — paydown progress (CC only)
    if (page.kind === "credit-card" && Number(page.credit_limit) > 0) {
      const limit = Number(page.credit_limit);
      const paidPct = Math.round((1 - balance / limit) * 100);
      const b3 = root.createEl("div", { cls: "dbt-band-3" });
      const bar = b3.createEl("div", { cls: "dbt-progress-bar" });
      bar.style.background = "#e5e7eb";
      const fill = bar.createEl("div", { cls: "dbt-progress-fill" });
      fill.style.width = `${paidPct}%`;
      fill.style.background = paidPct < 33 ? "#dc2626" : paidPct < 66 ? "#b45309" : "#16a34a";
      b3.createEl("div", { cls: "dbt-stat", text: `${paidPct}% paid down · Utilization ${100 - paidPct}%` });
    }
  }

  _renderSparkline(parent, values) {
    if (values.length === 0) return;
    const w = 200, h = 40;
    const min = Math.min(...values);
    const max = Math.max(...values);
    const range = max - min || 1;
    const pts = values.map((v, i) => `${(i / (values.length - 1)) * w},${h - ((v - min) / range) * h}`).join(" ");
    const svg = parent.createEl("div");
    svg.innerHTML = `<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" style="vertical-align:middle;"><polyline points="${pts}" fill="none" stroke="#6b7280" stroke-width="1.5"/></svg>`;
  }
}

    const totalInterest = debts.reduce((s, d) => s + ((Number(d.current_balance) || 0) * (Number(d.apr) || 0) / 100 / 12), 0);
    const totalPlanned = debts.reduce((s, d) => s + (Number(d.planned_monthly_payment) || 0), 0);
    const wAvgApr = totalBal > 0 ? debts.reduce((s, d) => s + (Number(d.current_balance) || 0) * (Number(d.apr) || 0), 0) / totalBal : 0;

    let zeroDate = "—";
    if (totalPlanned > totalInterest) {
      const principal = totalPlanned - totalInterest;
      const months = Math.ceil(totalBal / principal);
      const d = new Date();
      d.setMonth(d.getMonth() + months);
      zeroDate = d.toISOString().slice(0, 10);
    }

    const b1 = root.createEl("div", { cls: "dhs-band-1" });
    b1.createEl("div", { cls: "dhs-stat", text: `Total debt: $${totalBal.toFixed(2)}` });
    b1.createEl("div", { cls: "dhs-stat", text: `Monthly interest: $${totalInterest.toFixed(2)}` });
    b1.createEl("div", { cls: "dhs-stat", text: `Monthly attack: $${totalPlanned.toFixed(2)}` });
    b1.createEl("div", { cls: "dhs-stat", text: `Weighted APR: ${wAvgApr.toFixed(2)}%` });
    b1.createEl("div", { cls: "dhs-stat", text: `Zero-debt date: ${zeroDate}` });

    // Band 2 — per-kind strip
    const byKind = {};
    for (const d of debts) {
      const k = d.kind || "other";
      if (!byKind[k]) byKind[k] = { count: 0, total: 0 };
      byKind[k].count++;
      byKind[k].total += Number(d.current_balance) || 0;
    }
    const b2 = root.createEl("div", { cls: "dhs-band-2" });
    for (const [k, stats] of Object.entries(byKind)) {
      b2.createEl("span", { cls: "dhs-kind-chip", text: `${k}: $${stats.total.toFixed(0)} across ${stats.count}` });
    }

    // Band 3 — per-debt mini-cards sorted by balance DESC
    const b3 = root.createEl("div", { cls: "dhs-band-3" });
    const sorted = [...debts].sort((a, b) => (Number(b.current_balance) || 0) - (Number(a.current_balance) || 0));
    for (const d of sorted) {
      const card = b3.createEl("div", { cls: "dhs-card" });
      card.createEl("div", { cls: "dhs-card-name", text: d.name || "(unnamed)" });
      card.createEl("div", { cls: "dhs-card-bal", text: `$${(Number(d.current_balance) || 0).toFixed(2)}` });
      const apr = Number(d.apr) || 0;
      card.createEl("div", { cls: "dhs-card-apr", text: `${apr.toFixed(2)}% APR` });
      if (d.kind === "credit-card" && Number(d.credit_limit) > 0) {
        const paidPct = Math.round((1 - (Number(d.current_balance) || 0) / Number(d.credit_limit)) * 100);
        const chip = card.createEl("span", { cls: "dhs-card-chip", text: `${paidPct}% paid` });
        chip.style.background = paidPct < 33 ? "#dc2626" : paidPct < 66 ? "#b45309" : "#16a34a";
        chip.style.color = "#fff";
      }
      card.addEventListener("click", () => app.workspace.openLinkText(d.file.name, "", false));
      card.style.cursor = "pointer";
    }
  }
}
