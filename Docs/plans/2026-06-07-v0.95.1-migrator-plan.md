```js
// platform/blueprints/cowork/helpers/migrate-vault-config-helper.js — v0.95.1
//
// Pure helper. No I/O. Caller passes file content as string; helper returns
// modified string + per-engagement report. Caller writes .bak + modified.
//
// Inserts a compact `  overrides: {}` block into every engagement record in
// spice/cowork/context/vault-config.md that doesn't already carry one. Skips
// engagements with EITHER compact `overrides: {}` OR multi-line `overrides:`
// block-headers. Idempotent: re-runs against migrated content are no-ops.

"use strict";

function _findFrontmatterBoundaries(content) {
    // Matches `---\n<frontmatter>\n---` at start of file
    const m = content.match(/^---\n([\s\S]*?)\n---/);
    if (!m) return null;
    return { start: 0, fmStart: 4, fmEnd: 4 + m[1].length, end: 4 + m[1].length + 5 };
}

function _findEngagementBlocks(frontmatterText) {
    // Returns array of { id, blockStart, blockEnd } indices INTO the
    // frontmatter text (NOT the whole file). blockEnd is exclusive.
    const lines = frontmatterText.split("\n");
    let i = 0;
    // Locate engagements: line
    while (i < lines.length && !/^engagements:\s*$/.test(lines[i])) i++;
    if (i >= lines.length) return null; // signals "no engagements: key"
    i++;
    const blocks = [];
    // offset tracker: char offset of lines[i] within frontmatterText
    const lineOffsets = [0];
    for (let k = 0; k < lines.length - 1; k++) {
        lineOffsets.push(lineOffsets[k] + lines[k].length + 1); // +1 for \n
    }
    while (i < lines.length) {
        const m = lines[i].match(/^  -\s+id:\s*(.*)$/);
        if (m) {
            const id = m[1].trim().replace(/^["']|["']$/g, "");
            const startLine = i;
            let j = i + 1;
            while (j < lines.length) {
                const ln = lines[j];
                if (/^  -\s/.test(ln)) break;
                if (/^\S/.test(ln) && ln.trim() !== "") break;
                j++;
            }
            blocks.push({
                id,
                startLine,
                endLine: j,
                blockStart: lineOffsets[startLine],
                blockEnd: j < lines.length ? lineOffsets[j] : frontmatterText.length + 1,
            });
            i = j;
            continue;
        }
        if (/^\S/.test(lines[i]) && lines[i].trim() !== "") break; // left engagements list
        i++;
    }
    return blocks;
}

function _hasOverridesKey(blockText) {
    // Any line at field-indent (4 spaces) with `overrides:` key, compact or block-header
    return /^    overrides:/m.test(blockText);
}

function _detectIndentation(blockText) {
    // The engagement block uses 4-space field indent per existing schema; pinning
    // to that until a real-world variation surfaces.
    return "    ";
}

function _spliceOverridesBlock(blockText, indent) {
    // Insert `<indent>overrides: {}\n` immediately after the last non-blank,
    // non-comment line at field-indent level.
    const lines = blockText.split("\n");
    let insertAfter = -1;
    for (let i = lines.length - 1; i >= 0; i--) {
        const ln = lines[i];
        if (!ln.trim()) continue;
        if (ln.trim().startsWith("#")) continue;
        // Must be at field-indent level (starts with the indent string + non-space)
        if (ln.startsWith(indent) && !ln.startsWith(indent + " ")) {
            insertAfter = i;
            break;
        }
    }
    if (insertAfter < 0) {
        // Fall back to inserting at the end of the block (defensive — engagement
        // record had no field-level lines other than `- id:`)
        insertAfter = lines.length - 1;
        // Edge case: if block ends with trailing blank line(s), insert before them
        while (insertAfter > 0 && lines[insertAfter].trim() === "") insertAfter--;
    }
    const newLine = `${indent}overrides: {}`;
    lines.splice(insertAfter + 1, 0, newLine);
    return lines.join("\n");
}

function addEmptyOverridesBlock(content) {
    const errors = [];
    const changed = [];
    const skipped = [];

    if (typeof content !== "string" || content.length === 0) {
        return { modified: content || "", changed, skipped, errors: ["empty or non-string content"] };
    }

    const bounds = _findFrontmatterBoundaries(content);
    if (!bounds) {
        errors.push("frontmatter fences not found (missing leading `---\\n...---`)");
        return { modified: content, changed, skipped, errors };
    }

    const fmText = content.slice(bounds.fmStart, bounds.fmEnd);
    const blocks = _findEngagementBlocks(fmText);
    if (blocks === null) {
        errors.push("engagements: key not found in frontmatter");
        return { modified: content, changed, skipped, errors };
    }
    if (blocks.length === 0) {
        // engagements: present but empty — nothing to do
        return { modified: content, changed, skipped, errors };
    }

    // Walk blocks in REVERSE so we can splice without invalidating earlier offsets
    let newFmText = fmText;
    for (let i = blocks.length - 1; i >= 0; i--) {
        const blk = blocks[i];
        const blockText = newFmText.slice(blk.blockStart, blk.blockEnd);
        if (_hasOverridesKey(blockText)) {
            skipped.unshift(blk.id);
            continue;
        }
        const indent = _detectIndentation(blockText);
        const splicedBlock = _spliceOverridesBlock(blockText, indent);
        newFmText = newFmText.slice(0, blk.blockStart) + splicedBlock + newFmText.slice(blk.blockEnd);
        changed.unshift(blk.id);
    }

    const modified = content.slice(0, bounds.fmStart) + newFmText + content.slice(bounds.fmEnd);
    return { modified, changed, skipped, errors };
}

module.exports = {
    addEmptyOverridesBlock,
    // module-private (test access):
    _findFrontmatterBoundaries,
    _findEngagementBlocks,
    _hasOverridesKey,
    _spliceOverridesBlock,
    _detectIndentation,
};
```

```js
// platform/cli/cmd-migrate-config.js — `sauce migrate-config` verb (v0.95.1).
//
// Inserts an empty `overrides: {}` block into every engagement record in
// spice/cowork/context/vault-config.md that doesn't already carry one.
// Closes the v0.95.0 backward-compat one-cycle window for the

//
// Default mode is dry-run → reports what would change without writing.
// --apply writes a .bak sidecar (landmine #12) then overwrites the file.

"use strict";

const fs = require("fs");
const path = require("path");
const os = require("os");
const registry = require("./registry.js");

// Test-hook seam (mirrors cmd-bootstrap / cmd-migrate-frontmatter).
exports._helperImpl = null;

function expandTilde(p) {
    if (!p) return p;
    if (p === "~" || p === "~/") return process.env.HOME || os.homedir();
    if (p.startsWith("~/")) return path.join(process.env.HOME || os.homedir(), p.slice(2));
    return p;
}

exports._parseFlags = function(argv) {
    const flags = { vault: null, apply: false, help: false };
    for (let i = 0; i < (argv || []).length; i++) {
        const a = argv[i];
        if (a === "--help" || a === "-h") flags.help = true;
        else if (a === "--vault") flags.vault = expandTilde(argv[++i]);
        else if (a.startsWith("--vault=")) flags.vault = expandTilde(a.slice("--vault=".length));
        else if (a === "--apply") flags.apply = true;
        else throw new Error(`unknown arg: ${a}`);
    }
    return flags;
};

function resolveTargetVault(opts) {
    if (opts.vault) return path.resolve(opts.vault);
    const vaults = registry.list();
    if (vaults.length === 1) return path.resolve(vaults[0].path);
    if (vaults.length === 0) {
        throw new Error("no registered vaults; pass --vault <path>");
    }
    throw new Error("multiple registered vaults; pass --vault <path> to disambiguate");
}

exports.run = async function(ctx, args) {
    let flags;
    try { flags = exports._parseFlags(args || []); }
    catch (e) { console.error(`error: ${e.message}`); process.exit(2); return; }

    if (flags.help) {
        console.log(
            "Usage: sauce migrate-config [--vault <path>] [--apply]\n\n" +
            "Insert `overrides: {}` into every engagement record in vault-config.md\n" +
            "that doesn't already carry one. Default mode is dry-run; --apply writes.\n"
        );
        return;
    }

    let vaultRoot;
    try { vaultRoot = resolveTargetVault(flags); }
    catch (e) { console.error(`error: ${e.message}`); process.exit(2); return; }

    const target = path.join(vaultRoot, "spice/cowork/context/vault-config.md");
    if (!fs.existsSync(target)) {
        console.error(`error: no vault-config.md at ${target}`);
        process.exit(2);
        return;
    }

    const content = fs.readFileSync(target, "utf8");
    const helper = exports._helperImpl
        || require(path.join(vaultRoot, "spice/cowork/helpers/migrate-vault-config-helper.js"));
    const result = helper.addEmptyOverridesBlock(content);

    if (result.errors.length > 0) {
        console.error(`migrate-config aborted: ${result.errors.join("; ")}`);
        process.exit(2);
        return;
    }

    const changedSummary = result.changed.length === 0
        ? "(none)" : `${result.changed.length} (${result.changed.join(", ")})`;
    const skippedSummary = result.skipped.length === 0
        ? "(none)" : `${result.skipped.length} (${result.skipped.join(", ")})`;

    if (!flags.apply) {
        console.log(`[dry-run] would change: ${changedSummary}`);
        console.log(`[dry-run] would skip:   ${skippedSummary}`);
        console.log(`re-run with --apply to write changes.`);
        return;
    }

    if (result.changed.length === 0) {
        console.log(`migrate-config: no-op (all ${result.skipped.length} engagement(s) already have overrides:)`);
        return;
    }

    fs.copyFileSync(target, `${target}.bak`);
    fs.writeFileSync(target, result.modified);
    console.log(`migrate-config applied: changed ${changedSummary}; skipped ${skippedSummary}`);
    console.log(`backup: ${target}.bak`);
};
```
