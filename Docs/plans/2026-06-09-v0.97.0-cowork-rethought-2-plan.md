# v0.97.0 cowork-rethought-2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Confront v0.96.x's architectural mismatch — orchestrator SKILL.md bodies don't load in claude.ai's Cowork UI cron — by relocating orchestrator instructions into the wrapper template prompt body (Rail O+T), emitting sidecars via Obsidian MCP (Rail S), pushing wrapper changes via `update_scheduled_task` (Rail A), and running learning/heartbeat reconciliation via a local launchd nightly job (Rail R).

**Architecture:** Single-source `data/orchestrator-instructions/<cadence>.md` per cadence + `_shared-clauses.md`. SKILL.md becomes a 5-line shim. compose-helper substitutes `{{$tokens}}` + `{{shared.<key>}}` into wrapper template at sync time. Cron LLM resolves `{{$today_*}}` at fire time (kills timestamp-drift class). New `sauce reconcile-cowork` CLI verb fires nightly via launchd at 03:00 MT, processes both vaults via shared filesystem.

**Tech Stack:** Node 18+, claude.ai Cowork UI scheduled-task MCP (`list_scheduled_tasks`, `update_scheduled_task`, `create_scheduled_task`), launchd plist. Existing v0.96.x helpers (`learn-from-checks-helper.js`, `check-heartbeat-helper.js`, `validate-sidecar-helper.js`) called by Rail R reconciler. Existing harnesses `run-helper-cases.js` + `run-cowork-smoke.js` gain HC-V0970-O/T/S/A/R-* groups.

**Spec:** `Docs/plans/2026-06-09-v0.97.0-cowork-rethought-2-design.md`

function _loadOrchestratorInstructions(cadence, sourceRoot) {
  // sourceRoot defaults to workshop blueprint path; consumer-vault path also supported.
  const candidates = [
    sourceRoot && path.join(sourceRoot, `${cadence}.md`),
    path.join(__dirname, "..", "content", "data", "orchestrator-instructions", `${cadence}.md`),
  ].filter(Boolean);
  for (const p of candidates) {
    if (fs.existsSync(p)) return fs.readFileSync(p, "utf8");
  }
  throw new Error(`orchestrator-instructions file not found for cadence "${cadence}"`);
}

function _loadSharedClauses(sourceRoot) {
  const candidates = [
    sourceRoot && path.join(sourceRoot, "_shared-clauses.md"),
    path.join(__dirname, "..", "content", "data", "orchestrator-instructions", "_shared-clauses.md"),
  ].filter(Boolean);
  for (const p of candidates) {
    if (fs.existsSync(p)) {
      const raw = fs.readFileSync(p, "utf8");
      // Parse ## <key> blocks into a map
      const blocks = {};
      const sectionRx = /^## (\w+)\s*$/gm;
      const sections = [...raw.matchAll(sectionRx)];
      for (let i = 0; i < sections.length; i++) {
        const key = sections[i][1];
        const start = sections[i].index + sections[i][0].length;
        const end = i + 1 < sections.length ? sections[i + 1].index : raw.length;
        blocks[key] = raw.slice(start, end).trim();
      }
      return blocks;
    }
  }
  throw new Error(`_shared-clauses.md not found in ${candidates.join(" or ")}`);
}

function _resolveConditional(template, cadence_mode) {
  // {{#if cadence_mode == "lens_shift"}}...{{else}}...{{/if}}
  const rx = /\{\{#if cadence_mode == "([^"]+)"\}\}([\s\S]*?)\{\{else\}\}([\s\S]*?)\{\{\/if\}\}/g;
  return template.replace(rx, (_m, condValue, branchTrue, branchFalse) => {
    return cadence_mode === condValue ? branchTrue : branchFalse;
  });
}

function _substituteShared(template, sharedBlocks) {
  return template.replace(/\{\{shared\.(\w+)\}\}/g, (_m, key) => {
    if (sharedBlocks[key] == null) {
      throw new Error(`unknown shared key {{shared.${key}}}`);
    }
    return sharedBlocks[key];
  });
}

function _substituteStaticTokens(template, tokens) {
  return template.replace(/\{\{\$(\w+)\}\}/g, (m, key) => {
    // today_* tokens stay literal — return match unchanged
    if (key.startsWith("today_")) return m;
    // computed-at-LLM-time tokens stay literal
    if (key === "rating_kind_lines" || key === "pending_confirmation_lines") return m;
    if (tokens[key] == null) {
      // Unrecognized — leave literal for now; harness asserts no orphan {{$}} other than today_*
      return m;
    }
    return tokens[key];
  });
}

  // live:     Array<{ task_id, name, prompt, cron, enabled }>
  // Returns:  { changed, new: [], orphan: [], noop }
  if (!Array.isArray(composed)) throw new Error("composed must be an array");
  if (!Array.isArray(live)) throw new Error("live must be an array");

  const composedByName = new Map(composed.map((c) => [c.task_name, c]));
  const liveByName = new Map(live.map((l) => [l.name, l]));

  const result = { changed: [], new: [], orphan: [], noop: [] };

  for (const c of composed) {
    const liveTask = liveByName.get(c.task_name);
    if (!liveTask) {
      result.new.push({ task_name: c.task_name, composed_prompt: c.prompt });
    } else if (liveTask.prompt !== c.prompt) {
      result.changed.push({
        task_id: liveTask.task_id,
        task_name: c.task_name,
        composed_prompt: c.prompt,
        live_prompt_excerpt: liveTask.prompt.slice(0, 200),
      });
    } else {
      result.noop.push({ task_id: liveTask.task_id, task_name: c.task_name });
    }
  }

  for (const l of live) {
    if (!TASK_NAME_RX.test(l.name)) continue;  // Not a cowork-managed task
    if (!composedByName.has(l.name)) {
      result.orphan.push({
        task_id: l.task_id,
        task_name: l.name,
        live_cron: l.cron,
      });
    }
  }

  return result;
}

const fs = require("node:fs");
const path = require("node:path");
const os = require("node:os");

const REGISTRY_DIR = path.join(os.homedir(), ".sauce");
const REGISTRY_PATH = path.join(REGISTRY_DIR, "vault-paths.json");

function _ensureDir() {
  fs.mkdirSync(REGISTRY_DIR, { recursive: true });
}

function loadVaultPaths(opts) {
  const registryPath = (opts && opts.path) || REGISTRY_PATH;
  if (!fs.existsSync(registryPath)) {
    return { schema_version: "1.0.0", vaults: [] };
  }
  try {
    const raw = fs.readFileSync(registryPath, "utf8");
    const parsed = JSON.parse(raw);
    return {
      schema_version: parsed.schema_version || "1.0.0",
      vaults: Array.isArray(parsed.vaults) ? parsed.vaults : [],
    };
  } catch (err) {
    return { schema_version: "1.0.0", vaults: [], error: err.message };
  }
}

function addVaultPath({ path: vaultPath, label }, opts) {
  if (!vaultPath) throw new Error("addVaultPath: path required");
  const registryPath = (opts && opts.path) || REGISTRY_PATH;
  _ensureDir();
  const registry = loadVaultPaths({ path: registryPath });
  const exists = registry.vaults.find((v) => v.path === vaultPath);
  if (exists) {
    if (label && exists.label !== label) {
      exists.label = label;
    }
    fs.writeFileSync(registryPath, JSON.stringify(registry, null, 2) + "\n", "utf8");
    return { added: false, updated: !!label, registry };
  }
  registry.vaults.push({ path: vaultPath, label: label || path.basename(vaultPath) });
  fs.writeFileSync(registryPath, JSON.stringify(registry, null, 2) + "\n", "utf8");
  return { added: true, updated: false, registry };
}

const fs = require("node:fs");
const path = require("node:path");

function _readVaultConfig(vault_path) {
  const configPath = path.join(vault_path, "spice/cowork/context/vault-config.md");
  if (!fs.existsSync(configPath)) {
    throw new Error(`vault-config.md not found at ${configPath}`);
  }
  const raw = fs.readFileSync(configPath, "utf8");
  // Parse frontmatter — extract engagements[]
  const fmMatch = raw.match(/^---\s*\n([\s\S]+?)\n---/);
  if (!fmMatch) return { engagements: [] };
  return _parseEngagementsFromFrontmatter(fmMatch[1]);
}

function _parseEngagementsFromFrontmatter(fm) {
  // Minimal YAML parser for engagements: list
  const engagements = [];
  const lines = fm.split("\n");
  let inEngagements = false;
  let currentEng = null;
  for (const line of lines) {
    if (line.match(/^engagements:\s*$/)) {
      inEngagements = true;
      continue;
    }
    if (inEngagements && line.match(/^[a-z_]+:/)) {
      // Top-level key after engagements list — break
      if (line.match(/^  /)) continue;  // Indented = still in engagements
      inEngagements = false;
      break;
    }
    if (inEngagements) {
      const newEngMatch = line.match(/^\s*- id:\s*(.+)$/);
      if (newEngMatch) {
        if (currentEng) engagements.push(currentEng);
        currentEng = { id: newEngMatch[1].trim() };
        continue;
      }
      const fieldMatch = line.match(/^\s+([a-z_]+):\s*(.+)$/);
      if (fieldMatch && currentEng) {
        const key = fieldMatch[1];
        const val = fieldMatch[2].trim().replace(/^"(.*)"$/, "$1");
        currentEng[key] = val;
      }
    }
  }
  if (currentEng) engagements.push(currentEng);
  return { engagements };
}

async function reconcileVault({ vault_path, dry_run = false, engagement_filter = null }) {
  const errors = [];
  const summary = { vault_path, engagements_processed: [], dry_run };

  // Step 1: Read vault-config → engagements
  let config;
  try {
    config = _readVaultConfig(vault_path);
  } catch (err) {
    return { errors: [err.message], summary };
  }

  const engagements = engagement_filter
    ? config.engagements.filter((e) => e.id === engagement_filter)
    : config.engagements;

  for (const engagement of engagements) {
    const engSummary = { engagement_id: engagement.id, sidecars_backfilled: 0, sidecars_validated: 0, learned_weights_delta: {}, heartbeat: null };
    try {
      // Step 2: Process atomic notes
      const sidecarBackfill = require("./sidecar-backfill");
      const backfillResult = await sidecarBackfill.backfillMissing({ vault_path, engagement_id: engagement.id, dry_run });
      engSummary.sidecars_backfilled = backfillResult.backfilled_count;
      engSummary.sidecars_validated = backfillResult.validated_count;

      // Step 3: Update learned_weights
      const lwUpdate = require("./learned-weights-update");
      const lwResult = await lwUpdate.updateForEngagement({ vault_path, engagement_id: engagement.id, dry_run });
      engSummary.learned_weights_delta = lwResult.delta;

      // Step 4: Run check-heartbeat
      const heartbeatCheck = require("./heartbeat-check");
      const hbResult = await heartbeatCheck.runForEngagement({ vault_path, engagement_id: engagement.id, engagement, dry_run });
      engSummary.heartbeat = hbResult;
    } catch (err) {
      errors.push(`engagement ${engagement.id}: ${err.message}`);
    }
    summary.engagements_processed.push(engSummary);
  }

  // Step 5: Reconciler-log append
  if (!dry_run) {
    try {
      const reconcilerLog = require("./reconciler-log");

    } catch (err) {

    }
  }

  return { errors, summary };
}

const fs = require("node:fs");

const chb = require("../../blueprints/cowork/helpers/check-heartbeat-helper");

async function runForEngagement({ vault_path, engagement_id, engagement, dry_run = false }) {
  const enabled_cadences = [];
  if (engagement.cadences) {
    if (engagement.cadences.morning) enabled_cadences.push("morning-briefing");

    if (engagement.cadences.lens_shift) enabled_cadences.push("lens_shift");

  if (enabled_cadences.length === 0) {
    return { missed_count: 0, first_fire: true, skipped: "no-enabled-cadences" };
  }

  const last_fires = chb.walkCadenceSidecars({ vault_root: vault_path, engagement_id, days: 30 });
  const windows = chb._loadFreshnessWindows(vault_path).windows_hours;
  const result = chb.evaluateFreshness({ windows_hours: windows, expected_cadences: enabled_cadences, last_fires, now: new Date() });

  if (result.missed.length === 0) {
    return { missed_count: 0, first_fire: result.first_fire };
  }

  // Append warning callout to today's memory.md
  const today = new Date().toISOString().slice(0, 10);
  const calloutBody = chb.composeHeartbeatCallout(result.missed, result.first_fire, today);

  if (!dry_run) {
    const todayDirPath = _todayDirPath();
    const memoryPath = path.join(vault_path, "spice/cowork/memory", engagement_id, todayDirPath, "memory.md");
    if (fs.existsSync(memoryPath)) {
      const existing = fs.readFileSync(memoryPath, "utf8");
      // Idempotency: replace prior > [!warning]+ Cron heartbeat anomaly callout if present
      const newContent = _upsertHeartbeatCallout(existing, calloutBody);
      fs.writeFileSync(memoryPath, newContent, "utf8");
    }
  }

  return { missed_count: result.missed.length, first_fire: result.first_fire, missed: result.missed };
}

function _todayDirPath() {
  const now = new Date();
  const yyyy = now.getFullYear();
  const mm = String(now.getMonth() + 1).padStart(2, "0");

function _upsertHeartbeatCallout(existing, newCallout) {
  const calloutRx = /> \[!warning\]\+ Cron heartbeat anomaly[\s\S]*?(?=\n>?\s*\n[^>]|\n##|$)/;

  return existing.trimEnd() + "\n\n" + newCallout + "\n";

// platform/mechanisms/cowork-reconciler/sidecar-backfill.js

const fs = require("node:fs");
const path = require("node:path");

const RECONCILER_VERSION = "0.97.0";

const CADENCE_BY_FRONTMATTER_TYPE = {
  "cowork-morning-briefing": "morning-briefing",
  "cowork-morning-briefing-cold": "morning-briefing",
  "cowork-midday-tripwire": "midday-tripwire",
  "cowork-eod-review": "eod-review",
  "cowork-weekly-review": "weekly-review",
  "cowork-monthly-review": "monthly-review",
};

function _parseMdFrontmatter(content) {
  const m = content.match(/^---\s*\n([\s\S]+?)\n---/);
  if (!m) return { frontmatter: {}, body: content };
  const fm = {};
  for (const line of m[1].split("\n")) {
    const fieldMatch = line.match(/^([a-z_]+):\s*(.+)$/);
    if (fieldMatch) {
      const key = fieldMatch[1];
      const val = fieldMatch[2].trim().replace(/^"(.*)"$/, "$1");
      fm[key] = val;
    }
  }
  return { frontmatter: fm, body: content.slice(m[0].length) };
}

function _inferKindFromCalloutTitle(title, priorities) {
  const lower = title.toLowerCase();
  // Direct match: title starts with a kind name
  for (const kind of priorities) {
    if (lower.startsWith(kind.toLowerCase())) return kind;
  }
  // Heuristic substring matches
  if (lower.includes("calendar")) return "calendar";
  if (lower.includes("email") || lower.includes("gmail") || lower.includes("inbox")) return "email";
  if (lower.includes("chat") || lower.includes("teams") || lower.includes("slack") || lower.includes("conversation")) return "chat";
  if (lower.includes("message")) return "messages";
  if (lower.includes("finance") || lower.includes("debt") || lower.includes("scoreboard")) return "finance";
  if (lower.includes("project") || lower.includes("kanban")) return "projects";
  if (lower.includes("thread")) return "threads";
  if (lower.includes("github") || lower.includes("pull request")) return "github";
  if (lower.includes("ado") || lower.includes("azure devops") || lower.includes("work item")) return "ado";
  return null;
}

function backfillSidecar({ mdPath, engagementPriorities }) {
  const fileMtime = fs.statSync(mdPath).mtime;
  const content = fs.readFileSync(mdPath, "utf8");
  const { frontmatter } = _parseMdFrontmatter(content);
  const surfaced_kinds = [];
  const calloutRx = /^> \[!(?:example|warning)\][+\-]?\s+(.+)$/gm;
  let m;
  while ((m = calloutRx.exec(content))) {
    const title = m[1].trim();
    const kind = _inferKindFromCalloutTitle(title, engagementPriorities || []);
    if (kind && !surfaced_kinds.includes(kind)) surfaced_kinds.push(kind);
  }

  return {
    schema_version: "1.0.0",
    generated_by: `${frontmatter.generator || "cowork:unknown@?"} (reconciler-backfill@${RECONCILER_VERSION})`,
    generated_at: fileMtime.toISOString(),
    cadence: CADENCE_BY_FRONTMATTER_TYPE[frontmatter.type] || "unknown",
    engagement_id: frontmatter.engagement_id || "",
    frontmatter,
    surfaced_kinds,
    surfaced_items: [],
    render_aspects_applied: [],
    memory_used: { yesterday_present: false, drift_warning_present: false, echoes_count: 0 },
    plan_dispatch: { mode: "reconciler-backfill", kinds_dispatched: surfaced_kinds.length, warnings_emitted: 0 },
  };
}

async function backfillMissing({ vault_path, engagement_id, dry_run = false }) {
  const dailyRoot = path.join(vault_path, "spice/cowork/daily");
  let backfilled_count = 0;
  let validated_count = 0;

  if (!fs.existsSync(dailyRoot)) {
    return { backfilled_count, validated_count };
  }

  // Walk last 30 days
  const now = new Date();
  for (let d = 0; d < 30; d++) {
    const date = new Date(now.getTime() - d * 24 * 60 * 60 * 1000);
    const yyyy = date.getFullYear();
    const mm = String(date.getMonth() + 1).padStart(2, "0");
    const dd = String(date.getDate()).padStart(2, "0");
    const monthName = date.toLocaleString("en-US", { month: "long" });
    const dayDir = path.join(dailyRoot, String(yyyy), `${mm}-${monthName}`, `${yyyy}-${mm}-${dd}`);
    if (!fs.existsSync(dayDir)) continue;

    const files = fs.readdirSync(dayDir).filter((f) => f.endsWith(".md") && !f.endsWith(".cowork.json"));
    for (const file of files) {
      const mdPath = path.join(dayDir, file);
      const sidecarPath = mdPath.replace(/\.md$/, ".cowork.json");

      const content = fs.readFileSync(mdPath, "utf8");
      const { frontmatter } = _parseMdFrontmatter(content);
      if (engagement_id && frontmatter.engagement_id && frontmatter.engagement_id !== engagement_id) continue;

      if (!fs.existsSync(sidecarPath)) {
        // Backfill
        const sidecar = backfillSidecar({ mdPath, engagementPriorities: [] });
        if (!dry_run) {
          fs.writeFileSync(sidecarPath, JSON.stringify(sidecar, null, 2) + "\n", "utf8");
        }
        backfilled_count += 1;
      } else {
        validated_count += 1;
      }
    }
  }

  return { backfilled_count, validated_count };
}

module.exports = { backfillSidecar, backfillMissing, _inferKindFromCalloutTitle, RECONCILER_VERSION };

// platform/mechanisms/cowork-reconciler/launchd-installer.js

const fs = require("node:fs");
const path = require("node:path");
const os = require("node:os");
const { execSync } = require("node:child_process");

const TEMPLATE_PATH = path.join(__dirname, "..", "..", "..", "Docs", "install", "cowork-reconciler-launchd.plist.template");

function _sauceBinPath() {
  try {
    return execSync("which sauce", { encoding: "utf8" }).trim();
  } catch (_) {
    return "/opt/homebrew/bin/sauce";  // brew default on Apple Silicon
  }
}

async function installLaunchd() {
  if (!fs.existsSync(TEMPLATE_PATH)) {
    console.error(`Template not found at ${TEMPLATE_PATH}`);
    return 1;
  }
  const template = fs.readFileSync(TEMPLATE_PATH, "utf8");
  const user = process.env.USER || os.userInfo().username;
  const home = os.homedir();
  const saucePath = _sauceBinPath();

  const populated = template
    .replaceAll("{{$user}}", user)
    .replaceAll("{{$home}}", home)
    .replaceAll("{{$sauce_path}}", saucePath);

  const plistPath = path.join(home, "Library/LaunchAgents", `com.${user}.cowork-reconciler.plist`);
  fs.mkdirSync(path.dirname(plistPath), { recursive: true });
  fs.writeFileSync(plistPath, populated, "utf8");
  console.log(`Wrote launchd plist: ${plistPath}`);

  // Unload prior version (if any) and load the new one
  try {
    execSync(`launchctl unload -w ${plistPath} 2>/dev/null || true`, { stdio: "inherit" });
  } catch (_) { /* fine */ }
  try {
    execSync(`launchctl load -w ${plistPath}`, { stdio: "inherit" });
    console.log(`Loaded launchd job: com.${user}.cowork-reconciler`);
  } catch (err) {
    console.error(`launchctl load failed: ${err.message}`);
    return 1;
  }

  console.log(`\nNext fire: 03:00 local time. Logs:`);
  console.log(`  ${home}/Library/Logs/cowork-reconciler.log`);
  console.log(`  ${home}/Library/Logs/cowork-reconciler.err`);
  console.log(`\nTo uninstall: launchctl unload -w ${plistPath} && rm ${plistPath}`);
  return 0;
}
