/**
 * discover-people-helper.js — v0.28.0 (sauce v0.90.0)
 *
 * Pure helpers backing cowork:discover-people. No MCP calls, no fs (in
 * the pure paths), no stdout. Parses microscope ## What matters prose,
 * people-aliases.md sibling tables, and vault-config.md frontmatter to
 * produce a candidate-people list with provenance + classification.
 *
 * Exports:
 *   parseInnerCircleFromMicroscope(body) → string[]
 *   parsePromotionRowsFromSibling(body) → {rows: [{canonical_name, section, status}], suppress_list: string[]}
 *   parseStakeholdersFromVaultConfig(body, engagement_id) → string[]
 *   parseCurrentInnerCircle(body, engagement_id) → string[]

 *   composeReviewTable(candidates_with_status) → string
 *   composeReport({applied, skipped, sources_scanned, engagement_id}) → string

// --- parseInnerCircleFromMicroscope -----------------------------------------

function parseInnerCircleFromMicroscope(body) {
    if (!body || typeof body !== "string") return [];
    const lines = body.split("\n");
    // Find the ## What matters section
    let inSection = false;
    let sectionLines = [];
    for (const line of lines) {
        if (/^##\s+What\s+matters\b/i.test(line)) { inSection = true; continue; }
        if (inSection && /^##\s+/.test(line)) break;
        if (inSection) sectionLines.push(line);
    }
    if (sectionLines.length === 0) return [];
    const text = sectionLines.join("\n");
    // Pattern: "Inner circle (people)...: A, B, C, ..."
    // And: "Elevation signals: ... Priority pair: A and B. C. ..."
    const out = new Set();

    // Extract from "Inner circle (people)..." lines
    const innerMatches = text.match(/Inner\s+circle\s*\(people\)[^:]*:\s*([^\n.]+)/i);
    if (innerMatches) {
        const namesPart = innerMatches[1];
        namesPart.split(/,|\sand\s/).forEach(n => {
            const cleaned = n.trim().replace(/^\s*resolved.*?:?\s*/i, "").trim();
            if (cleaned && /^[A-Z]/.test(cleaned)) out.add(cleaned);
        });
    }

    // Extract from "Priority pair: A and B" and standalone "Name." sentences inside Elevation signals
    const priorityMatches = text.match(/Priority\s+pair:\s*([^.\n]+)/i);
    if (priorityMatches) {
        priorityMatches[1].split(/,|\sand\s/).forEach(n => {
            const cleaned = n.trim();
            if (cleaned && /^[A-Z]/.test(cleaned)) out.add(cleaned);
        });
    }

    // Bare name sentences in elevation list: lines starting with "- " and ending in "."
    const bullets = sectionLines.filter(l => /^\s*-\s+[A-Z]/.test(l));
    for (const b of bullets) {
        const m = b.match(/^\s*-\s+([A-Z][a-zA-Z]+(?:\s+[A-Z][a-zA-Z]+)*)\.?$/);
        if (m) out.add(m[1].trim());
    }

    return Array.from(out);
}

// --- parsePromotionRowsFromSibling ------------------------------------------

function parsePromotionRowsFromSibling(body) {
    const result = { rows: [], suppress_list: [] };
    if (!body || typeof body !== "string") return result;

    // Tokenize the body into sections by ## headings
    const sections = body.split(/^##\s+/m);
    for (const section of sections) {
        const firstLine = section.split("\n")[0] || "";
        const isInnerCircle = /inner\s+circle/i.test(firstLine) && !/observed|frequent/i.test(firstLine);
        const isFrequent = /frequent\s+collaborators|observed/i.test(firstLine);
        const isSuppress = /personal|non-work|suppress/i.test(firstLine);
        const isElevation = /elevation/i.test(firstLine) && !/inner/i.test(firstLine);

        if (!isInnerCircle && !isFrequent && !isSuppress && !isElevation) continue;
        // Find markdown tables in this section
        const tableRows = section.split("\n").filter(l => /^\|/.test(l));
        for (const row of tableRows) {
            if (/---/.test(row)) continue;  // separator row
            const cells = row.split("|").map(c => c.trim()).filter(c => c);
            if (cells.length < 2) continue;
            if (/^Name$|^Inner circle|^canonical|^Status$|^Promote/i.test(cells[0])) continue;  // header row
            // For suppress section, just collect names
            if (isSuppress) {
                if (cells[0]) result.suppress_list.push(cells[0]);
                continue;
            }
            // For inner_circle / frequent / elevation: canonical_name in column 2, status in last column
            const canonical = cells[1] || cells[0];
            const status = (cells[cells.length - 1] || "").toLowerCase();
            let normStatus = status;
            if (status.includes("confirm") && !status.includes("ed")) normStatus = "confirm";
            else if (status.includes("confirmed")) normStatus = "confirmed";
            else if (status.includes("promoted")) normStatus = "promoted";
            else if (status.includes("skip")) normStatus = "skip";
            const section_name = isInnerCircle ? "inner_circle"
                : isFrequent ? "frequent_collaborators"
                : isElevation ? "elevation"
                : "unknown";
            result.rows.push({ canonical_name: canonical, section: section_name, status: normStatus });
        }
    }
    return result;
}

// --- parseStakeholdersFromVaultConfig ---------------------------------------

function parseStakeholdersFromVaultConfig(body, engagement_id) {
    if (!body || !engagement_id) return [];
    // Extract frontmatter
    const m = body.match(/^---\s*\n([\s\S]*?)\n---\s*\n?/);
    if (!m) return [];
    const fm = m[1];
    // Walk to the engagement with matching id
    const engRegex = new RegExp(`-\\s+id:\\s+["']?${engagement_id}["']?[\\s\\S]*?(?=^\\s*-\\s+id:|^[a-z]|^---)`, "m");
    const engMatch = fm.match(engRegex);
    if (!engMatch) return [];
    const engBlock = engMatch[0];
    const names = new Set();
    // stakeholders array
    const stakeMatch = engBlock.match(/stakeholders:\s*\n((?:\s+-\s+.*\n?)+)/);
    if (stakeMatch) {
        stakeMatch[1].split("\n").forEach(l => {
            const nm = l.match(/^\s+-\s+["']?([^"'\n]+?)["']?\s*$/);
            if (nm && nm[1]) names.add(nm[1].trim());
        });
    }
    // manager
    const mgrMatch = engBlock.match(/manager:\s+["']?([^"'\n]+?)["']?\s*$/m);
    if (mgrMatch && mgrMatch[1]) names.add(mgrMatch[1].trim());
    // direct_reports array (if present)
    const drMatch = engBlock.match(/direct_reports:\s*\n((?:\s+-\s+.*\n?)+)/);
    if (drMatch) {
        drMatch[1].split("\n").forEach(l => {
            const nm = l.match(/^\s+-\s+["']?([^"'\n]+?)["']?\s*$/);
            if (nm && nm[1]) names.add(nm[1].trim());
        });
    }
    return Array.from(names);
}

// --- parseCurrentInnerCircle ------------------------------------------------

function parseCurrentInnerCircle(body, engagement_id) {
    if (!body || !engagement_id) return [];
    const m = body.match(/^---\s*\n([\s\S]*?)\n---\s*\n?/);
    if (!m) return [];
    const fm = m[1];
    const engRegex = new RegExp(`-\\s+id:\\s+["']?${engagement_id}["']?[\\s\\S]*?(?=^\\s*-\\s+id:|^[a-z]|^---)`, "m");
    const engMatch = fm.match(engRegex);
    if (!engMatch) return [];
    const innerMatch = engMatch[0].match(/inner_circle_people:\s*\n((?:\s+-\s+.*\n?)+)/);
    if (!innerMatch) return [];
    const out = [];
    innerMatch[1].split("\n").forEach(l => {
        const nm = l.match(/^\s+-\s+["']?([^"'\n]+?)["']?\s*$/);
        if (nm && nm[1]) out.push(nm[1].trim());
    });
    return out;
}

// --- aggregateCandidates ----------------------------------------------------

function aggregateCandidates({ microscopeNames, siblingRows, vaultConfigNames }) {
    const map = new Map();  // canonical_name (case-insensitive key) → candidate
    function _norm(s) { return String(s).trim().toLowerCase(); }
    function _add(name, source) {
        if (!name) return;
        const key = _norm(name);
        if (!map.has(key)) {
            map.set(key, { canonical_name: name, sources: [] });
        }
        const c = map.get(key);
        c.sources.push(source);
    }
    Object.keys(microscopeNames || {}).forEach(kind => {
        (microscopeNames[kind] || []).forEach(name => {
            _add(name, { source_type: "microscope", source_kind: kind, source_path: `spice/cowork/prompts/per-mcp/${kind}/microscope.md` });
        });
    });
    Object.keys(siblingRows || {}).forEach(kind => {
        (siblingRows[kind] || []).forEach(row => {
            _add(row.canonical_name, { source_type: "sibling", source_kind: kind, source_path: `spice/cowork/prompts/per-mcp/${kind}/people-aliases.md`, sibling_status: row.status, sibling_section: row.section });
        });
    });
    (vaultConfigNames || []).forEach(name => {
        _add(name, { source_type: "vault_config", source_path: "spice/cowork/context/vault-config.md" });
    });
    return Array.from(map.values());
}

// --- classifyCandidate ------------------------------------------------------

function classifyCandidate({ candidate, resolveResult, current_inner_circle, suppress_list }) {
    const inSuppress = (suppress_list || []).some(s => String(s).toLowerCase() === String(candidate.canonical_name).toLowerCase());
    if (inSuppress) return "SUPPRESS";
    const inInnerCircle = (current_inner_circle || []).some(s => String(s).toLowerCase() === String(candidate.canonical_name).toLowerCase());
    if (inInnerCircle) return "ALREADY_PROMOTED";
    // sibling_status "skip" honors prior user decision
    const skippedInSibling = (candidate.sources || []).some(s => s.sibling_status === "skip");
    if (skippedInSibling) return "SUPPRESS";
    if (resolveResult && resolveResult.resolved) return "PROMOTE_EXISTING";
    return "CREATE_AND_PROMOTE";
}

// --- composePersonNoteStub --------------------------------------------------

function composePersonNoteStub({ canonical_name, aliases, sources }) {
    const aliasArr = Array.isArray(aliases) ? aliases : [];
    const aliasYaml = aliasArr.length
        ? aliasArr.map(a => `  - {type: ${a.type}, value: ${JSON.stringify(a.value)}}`).join("\n")
        : "";
    const provLine = (sources || []).map(s => `${s.source_type}${s.source_kind ? "/" + s.source_kind : ""}`).join(", ");
    return [
        "---",
        `name: ${canonical_name}`,
        `aliases:${aliasArr.length ? "\n" + aliasYaml : " []"}`,
        `created_by: cowork:discover-people`,
        "---",
        "",
        `# ${canonical_name}`,
        "",
        `> [!info]- Discovered by cowork:discover-people`,
        `> Sources: ${provLine || "(none recorded)"}`,
        "",
    ].join("\n");
}

// --- composeUpdatedSibling --------------------------------------------------

function composeUpdatedSibling(body, name_status_map) {
    if (!body || typeof body !== "string") return body || "";
    if (!name_status_map || typeof name_status_map !== "object") return body;
    const lines = body.split("\n");
    const out = lines.map(line => {
        if (!/^\|/.test(line)) return line;
        const cells = line.split("|");
        // typical row format: ["", " name ", " canonical_name ", " where seen ", " status ", ""]
        // find canonical-name cell (column 2 for collaborators table; column 2 for inner-circle confirmed table)
        // Try both: column 1 OR 2 as the name token
        for (const name of Object.keys(name_status_map)) {
            if (cells.some(c => c.trim() === name)) {
                // Flip last non-empty cell (status column) to new value if it was "confirm"
                const lastIdx = cells.length - 2;  // last cell before trailing empty
                if (cells[lastIdx] && cells[lastIdx].trim() === "confirm") {
                    cells[lastIdx] = ` ${name_status_map[name]} `;
                    return cells.join("|");
                }
            }
        }
        return line;
    });
    return out.join("\n");
}

// --- composeUpdatedVaultConfig ----------------------------------------------

function composeUpdatedVaultConfig(body, engagement_id, names_to_append) {
    if (!body || !engagement_id || !Array.isArray(names_to_append)) return body || "";
    // Find the engagement block
    const fmMatch = body.match(/^(---\s*\n)([\s\S]*?)(\n---\s*\n?)/);
    if (!fmMatch) return body;
    const before = fmMatch[1];
    let fm = fmMatch[2];
    const after = fmMatch[3] + body.slice(fmMatch[0].length);

    const engRegex = new RegExp(`(  - id:\\s+["']?${engagement_id}["']?[\\s\\S]*?)(inner_circle_people:\\s*\\n)((?:    -\\s+.*\\n?)*)`, "m");
    const m = fm.match(engRegex);
    if (m) {
        const existing = m[3];
        const existingNames = new Set();
        existing.split("\n").forEach(l => {
            const nm = l.match(/^\s+-\s+["']?([^"'\n]+?)["']?\s*$/);
            if (nm && nm[1]) existingNames.add(nm[1].trim());
        });
        const appendLines = names_to_append
            .filter(n => !existingNames.has(n))
            .map(n => `    - "${n}"`)
            .join("\n");
        const newBlock = m[1] + m[2] + existing + (appendLines ? appendLines + "\n" : "");
        fm = fm.replace(m[0], newBlock);
    } else {
        // No inner_circle_people block exists for this engagement; insert one
        const engOnly = new RegExp(`(  - id:\\s+["']?${engagement_id}["']?[\\s\\S]*?)(?=\\n  - id:|\\nmcp_map:|$)`, "m");
        const em = fm.match(engOnly);
        if (em) {
            const insertion = `    inner_circle_people:\n` + names_to_append.map(n => `      - "${n}"`).join("\n") + "\n";
            fm = fm.replace(em[0], em[0] + insertion);
        }
    }
    return before + fm + after;
}

// --- composeReviewTable -----------------------------------------------------

function composeReviewTable(candidates_with_status) {
    const groups = { CREATE_AND_PROMOTE: [], PROMOTE_EXISTING: [], ALREADY_PROMOTED: [], SUPPRESS: [] };
    (candidates_with_status || []).forEach(c => {
        if (groups[c.status]) groups[c.status].push(c);
    });
    const lines = [];
    let rowNum = 1;
    function _fmtSources(sources) {
        return (sources || []).map(s => `${s.source_type}${s.source_kind ? ":" + s.source_kind : ""}`).join(", ");
    }
    Object.keys(groups).forEach(status => {
        if (groups[status].length === 0) return;
        lines.push(`### ${status} (${groups[status].length})`);
        lines.push("");
        lines.push("| # | Canonical Name | Sources |");
        lines.push("|---|---|---|");
        groups[status].forEach(c => {
            lines.push(`| ${rowNum++} | ${c.canonical_name} | ${_fmtSources(c.sources)} |`);
        });
        lines.push("");
    });
    return lines.join("\n");
}

// --- composeReport ----------------------------------------------------------

function composeReport({ applied, skipped, sources_scanned, engagement_id }) {
    const lines = [];
    lines.push(`# cowork:discover-people report — engagement \`${engagement_id || "(unknown)"}\``);
    lines.push("");
    lines.push("## Applied");
    lines.push("");
    if (applied && applied.length) applied.forEach(a => lines.push(`- **${a.canonical_name}** — ${a.action}`));
    else lines.push("(none)");
    lines.push("");
    lines.push("## Skipped");
    lines.push("");
    if (skipped && skipped.length) skipped.forEach(s => lines.push(`- **${s.canonical_name}** — ${s.reason}`));
    else lines.push("(none)");
    lines.push("");
    lines.push("## Sources scanned");
    lines.push("");
    if (sources_scanned && sources_scanned.length) sources_scanned.forEach(s => lines.push(`- ${s}`));
    else lines.push("(none)");
    return lines.join("\n");
}

# cowork:discover-people

Microscope-extraction first slice for populating the canonical `inner_circle_people` allowlist that v0.89.0+ gather skills + v0.89.1+ atomic-note orchestrators consume. Scans USER-AUTHORED microscope prose + sibling people-aliases tables + vault-config structured engagement fields, dedups, classifies via the v0.89.0 `cowork:resolve-person` sub-skill, presents a batch-confirm review table, and on confirmation writes person-note stubs + appends names to `inner_circle_people` + flips `Promote? confirm` columns to `promoted`.

This skill does NOT make MCP discovery calls in v0.28.0 — that's WS-C's v0.29.0+ extension. The skill reads what the user has already authored in microscope.md + sibling files + vault-config and operationalizes it.

## Inputs

```
{
  engagement_id?: string   // optional; default-if-one, ask-if-many
}
```

## Pre-flight

1. Use Skill `cowork:check-vault-routing` with `{ required: ["obsidian"] }`. If not `"ready"`, emit Notice `cowork:discover-people aborted -- obsidian MCP unavailable` and exit.

2. **Resolve target engagement.** Read `<vault>/spice/cowork/context/vault-config.md` via `mcp__obsidian__obsidian_get_file_contents`.

   - Parse frontmatter `engagements[]`.
   - If `engagement_id` input provided + valid → use it.
   - Else if `engagements.length === 1` → auto-pick (default-if-one).
   - Else (`engagements.length > 1`) → ask user one question listing options.
   - If engagement not found → emit Notice `cowork:discover-people aborted -- engagement <id> not found in vault-config.md` and exit.
   - Capture `engagement` + `engagement_id`.

## Discover

3. **Enumerate per-mcp directories.** Call `mcp__obsidian__obsidian_list_files_in_dir` on `spice/cowork/prompts/per-mcp/`. For each subdirectory (`<kind>`):

   3a. Try reading `spice/cowork/prompts/per-mcp/<kind>/microscope.md` via `mcp__obsidian__obsidian_get_file_contents`. Treat a not-found error as absent. Capture body as `microscopes[<kind>]`.

   3b. Try reading `spice/cowork/prompts/per-mcp/<kind>/people-aliases.md`. Treat not-found as absent. Capture body as `siblings[<kind>]`.

4. **Parse + aggregate.** Require the pure helper:

   ```js
   const DPH = require("<workshop>/platform/blueprints/cowork/helpers/discover-people-helper.js");
   ```

   For each kind:
   - `microscopeNames[<kind>] = DPH.parseInnerCircleFromMicroscope(microscopes[<kind>] || "")`
   - `siblingRows[<kind>] = DPH.parsePromotionRowsFromSibling(siblings[<kind>] || "").rows`
   - Accumulate `suppress_list` from each kind's sibling.

   Read vault-config.md (already read in pre-flight step 2).
   - `vaultConfigNames = DPH.parseStakeholdersFromVaultConfig(vaultConfigBody, engagement_id)`
   - `current_inner_circle = DPH.parseCurrentInnerCircle(vaultConfigBody, engagement_id)`

   Call `candidates = DPH.aggregateCandidates({ microscopeNames, siblingRows, vaultConfigNames })`.

## Classify

5. For each candidate, call sub-skill `cowork:resolve-person` with `{ input: candidate.canonical_name, prefer_type: "name", engagement_id }`. Capture `resolveResult`.

   Compute status: `DPH.classifyCandidate({ candidate, resolveResult, current_inner_circle, suppress_list })` → one of `CREATE_AND_PROMOTE`, `PROMOTE_EXISTING`, `ALREADY_PROMOTED`, `SUPPRESS`.

   Annotate each candidate with `status` + `resolveResult`.

## Review

6. **Compose review table** via `DPH.composeReviewTable(candidates_with_status)`. Print to the chat for user inspection.

7. **Ask user (single batch-confirm question):**
   - Tell the user: "Apply these N actions? Respond with `y` (apply all), `n` (apply none), or `skip <row1,row2,...>` to apply all except listed rows."
   - Parse response. On `n` → emit Notice `cowork:discover-people exited without writes` and stop. On `y` → all proposed actions apply. On `skip <list>` → drop named rows from the apply set.

## Apply

8. **Person-note stubs.** For each candidate with status `CREATE_AND_PROMOTE` in the apply set:
   - Pre-check existence via `mcp__obsidian__obsidian_get_file_contents` on `spice/people/<canonical_name>.md`. If exists, log Notice `cowork:discover-people: spice/people/<name>.md already exists; skipping stub creation` and skip this write (but still proceed with vault-config promotion).
   - If absent: call `DPH.composePersonNoteStub({ canonical_name, aliases: [], sources: candidate.sources })`. Write to `spice/people/<canonical_name>.md` via `mcp__obsidian__obsidian_put_content`.

9. **Vault-config promotion.** Collect `names_to_append` = canonical names of all `CREATE_AND_PROMOTE` + `PROMOTE_EXISTING` candidates in the apply set. Compose updated body:
   - Read `spice/cowork/context/vault-config.md`. Write `.bak` per landmine #12 backup convention (Bash: `cp <path> <path>.bak`).
   - `updated = DPH.composeUpdatedVaultConfig(body, engagement_id, names_to_append)`.
   - Write back via `mcp__obsidian__obsidian_put_content`.

10. **Sibling column flips.** For each kind where applied candidates had sibling-row provenance:
    - Read `spice/cowork/prompts/per-mcp/<kind>/people-aliases.md`. Write `.bak`.
    - Build `name_status_map = { <canonical_name>: "promoted" }` for applied candidates.
    - `updated = DPH.composeUpdatedSibling(body, name_status_map)`.
    - Write back via `mcp__obsidian__obsidian_put_content`.

## Report

11. **Compose final report** via `DPH.composeReport({ applied, skipped, sources_scanned, engagement_id })`. Print to chat. Emit final Notice: `cowork:discover-people complete -- <N applied> applied, <M skipped> skipped for engagement <id>`.

## Idempotence + re-run

- The helper's `composeUpdatedVaultConfig` dedups against existing `inner_circle_people` entries — re-runs do not duplicate.
- The helper's `composeUpdatedSibling` only flips `confirm → promoted`; `promoted` and `skip` rows are untouched.
- Person-note stubs are NOT created if the file already exists (pre-check at Step 8).
- Naive re-discovery: previously declined candidates re-appear in the next run's review table. User re-declines.

## Default-if-one engagement resolution

When `engagement_id` is omitted from inputs:
- 1 engagement in vault-config → auto-pick that one. Emit Notice `cowork:discover-people: defaulting to engagement <id> (single-engagement vault)`.
- More than 1 engagement → prompt user with multiple-choice listing labels + ids; require valid selection.

## Sibling-file safety (landmine #12 backup)

Both `vault-config.md` and `people-aliases.md` files are USER-OWNED in the cowork manifest (vault-config.md is materialized once; people-aliases.md is not in `files[]` at all — pure user creation via `cowork:edit-microscope`). Before any modification to either: Bash `cp <file> <file>.bak`. User can revert from `.bak` if they object to the changes. Per landmine #12, never write to allowlisted `.obsidian/` paths.

## Dependencies

- `cowork:check-vault-routing` (pre-flight Step 1) — required
- `cowork:resolve-person` (Step 5, per-candidate) — required

Helper: `platform/blueprints/cowork/helpers/discover-people-helper.js` (pure, materialized to `spice/cowork/helpers/discover-people-helper.js` on consumer vaults).

## Outputs

- Side-effect writes (Steps 8-10): new person-note stubs, updated vault-config, updated sibling files (with .bak backups).
- Stdout: review table (Step 6), report (Step 11).
- Notice: final completion confirmation.
