- [ ] **Step 1: Add `deriveExcludedThemes` to dispatch-plan-helper.js**

```javascript
/**
 * Derive excluded_themes[] from yesterday's memory carry-forward bullets.
 * Returns raw bullet strings (no extraction; compose-body LLM does semantic compare).
 * @param {Object | null} yesterdayMemory - return value of read-memory sub-skill
 * @returns {string[]} - array of bullet strings, [] if no memory or no bullets
 */
function deriveExcludedThemes(yesterdayMemory) {
  if (!yesterdayMemory) return [];
  const bullets = yesterdayMemory.carry_forward_bullets;
  if (!Array.isArray(bullets) || bullets.length === 0) return [];
  return bullets.filter(b => typeof b === 'string' && b.trim().length > 0);
}

module.exports = {
  // ...existing exports...
  deriveExcludedThemes,
};
```

- [ ] **Step 2: Add `injectAntiEchoCallout` to compose-body-helper.js**

```javascript
const ANTI_ECHO_ELIGIBLE_CADENCES = ['morning-briefing', 'midday-tripwire', 'eod-review'];

/**
 * Inject the anti-echo callout into a body string.
 * Caller is responsible for cadence-eligibility check OR call composeBody which
 * handles it; this helper assumes caller already gated.
 * @param {string} body - current body string
 * @param {string[]} excluded_themes - non-empty array of bullet strings
 * @param {Object} voice_contract - engagement voice contract
 * @returns {string} - body with callout appended near end (before tip-close section)
 */
function injectAntiEchoCallout(body, excluded_themes, voice_contract) {
  const joined = excluded_themes.join(' | ');
  const callout = `\n> [!question] Outside yesterday's frame\n> {{LLM-fill: ONE item from today's dispatch that doesn't relate to any of: ${joined}. If nothing qualifies, write 'today's gather largely continued yesterday's threads.' Apply voice contract.}}\n`;
  // Insert before the tip-close section if present, else append at end
  const tipCloseIdx = body.lastIndexOf('> [!tip]');
  if (tipCloseIdx > 0) {
    return body.slice(0, tipCloseIdx) + callout + '\n' + body.slice(tipCloseIdx);
  }
  return body + callout;
}

module.exports = {
  // ...existing exports...
  injectAntiEchoCallout,
  ANTI_ECHO_ELIGIBLE_CADENCES,
};
```

- [ ] **Step 2: Author CoworkLensShiftCards.js CustomJS class**

```javascript
class CoworkLensShiftCards {
  /**
   * Render warm/cold MB pairs for the current week as 2-column cards.
   * @param {object} dv - Dataview object provided by Obsidian
   * @param {object} input - {scope: "this_week" | "today" | "this_month"}
   */
  async render(dv, { scope = "this_week" } = {}) {
    const now = dv.luxon.DateTime.now();
    const weekStart = now.startOf('week');
    const weekEnd = now.endOf('week');

    // Find all warm + cold MB notes in the week
    const notes = dv.pages('"spice/cowork/daily"')
      .where(p => p.type === "cowork-morning-briefing" || p.type === "cowork-morning-briefing-cold")
      .where(p => {
        const created = dv.luxon.DateTime.fromISO(p.created_at);
        return created >= weekStart && created <= weekEnd;
      });

    // Group by (day, engagement_id), pair warm + cold
    const pairs = {};
    for (const n of notes) {
      const day = n.file.folder.split('/').pop(); // YYYY-MM-DD
      const eng = n.engagement_id;
      const key = `${day}::${eng}`;
      if (!pairs[key]) pairs[key] = { day, engagement_id: eng, warm: null, cold: null };
      if (n.type === "cowork-morning-briefing-cold") pairs[key].cold = n;
      else pairs[key].warm = n;
    }

    // Render
    if (Object.keys(pairs).length === 0) {
      dv.paragraph("_No lens-shift companions this week._");
      return;
    }

    for (const { day, engagement_id, warm, cold } of Object.values(pairs)) {
      dv.header(3, `${day} — ${engagement_id}`);
      const cols = [];
      if (warm) cols.push(`**Warm (read memory)**\n${warm.summary || ''}\n[Open →](${warm.file.path})`);
      if (cold) cols.push(`**Cold (no memory)**\n${cold.summary || ''}\n[Open →](${cold.file.path})`);
      dv.table(cols.length === 2 ? ['Warm', 'Cold'] : ['Note'], [cols]);
    }
  }
}
```
