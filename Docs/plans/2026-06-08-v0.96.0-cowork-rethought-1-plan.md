```javascript
// platform/blueprints/cowork/helpers/validate-sidecar-helper.js
//
// Validates a cowork .cowork.json sidecar against a cadence-specific JSON
// Schema (draft-07). Used by write-atomic-note-helper before write commits,
// and by the orchestrator Verify pass to replace v0.91.x–v0.92.0 regex
// markers. Returns { ok, errors? }.

const fs = require("node:fs");
const path = require("node:path");

let _ajv = null;
function _getAjv() {
  if (_ajv) return _ajv;
  // Lazy-load Ajv only when first validation runs. Cowork is the only
  // platform blueprint that depends on Ajv; lazy-load avoids paying the
  // require cost when other blueprints' tests run.
  const Ajv = require("ajv");
  _ajv = new Ajv({ allErrors: true, strict: false });
  return _ajv;
}

const _schemaCache = new Map();
function _loadSchema(schemaPath) {
  if (_schemaCache.has(schemaPath)) return _schemaCache.get(schemaPath);
  const raw = fs.readFileSync(schemaPath, "utf8");
  const schema = JSON.parse(raw);
  _schemaCache.set(schemaPath, schema);
  return schema;
}

function validateSidecar(payload, schemaPath) {
  if (!payload || typeof payload !== "object") {
    return { ok: false, errors: [{ message: "payload must be an object" }] };
  }
  if (!fs.existsSync(schemaPath)) {
    return { ok: false, errors: [{ message: `schema not found at ${schemaPath}` }] };
  }
  const schema = _loadSchema(schemaPath);
  const ajv = _getAjv();
  const validate = ajv.compile(schema);
  const ok = validate(payload);
  if (ok) {
    if (Array.isArray(payload.surfaced_items)) {
      const ids = new Set();
      for (const item of payload.surfaced_items) {
        if (ids.has(item.item_id)) {
          return {
            ok: false,
            errors: [{ message: `duplicate item_id within sidecar: ${item.item_id}` }],
          };
        }
        ids.add(item.item_id);
      }
    }
    return { ok: true };
  }
  return { ok: false, errors: validate.errors || [] };
}

function validatePreCycleNote(mdPath) {
  return { ok: true, warn: "missing-sidecar", mdPath };
}

module.exports = { validateSidecar, validatePreCycleNote };
```

- [ ] **Step 2: Implement `write-atomic-note-helper.js`**

```javascript
// platform/blueprints/cowork/helpers/write-atomic-note-helper.js
//
// Atomic dual-write for cowork atomic notes: writes the .md and its
// .cowork.json sidecar together; on any failure, BOTH files are deleted
// so the directory never carries a half-written pair. Sidecar is
// validated against its cadence-specific JSON Schema BEFORE write commits.
// Returns a status string matching the v0.91.x–v0.92.0 contract-violation
// convention.

const fs = require("node:fs");
const path = require("node:path");
const { validateSidecar } = require("./validate-sidecar-helper");

function _writeFileAtomic(filePath, contents) {
  const dir = path.dirname(filePath);
  fs.mkdirSync(dir, { recursive: true });
  const tmp = `${filePath}.tmp-${process.pid}-${Date.now()}`;
  fs.writeFileSync(tmp, contents, "utf8");
  fs.renameSync(tmp, filePath);
}

function _safeUnlink(filePath) {
  try { fs.unlinkSync(filePath); } catch (_) { /* ignore */ }
}

function writeAtomicNote({ mdPath, sidecarPath, body_md, sidecar_json, schemaPath }) {
  if (!mdPath || !sidecarPath || !body_md || !sidecar_json || !schemaPath) {
    return { status: "failed:contract-violation:missing-input" };
  }
  const validation = validateSidecar(sidecar_json, schemaPath);
  if (!validation.ok) {
    return {
      status: "failed:contract-violation:sidecar-schema",
      errors: validation.errors,
    };
  }
  try {
    _writeFileAtomic(mdPath, body_md);
  } catch (err) {
    return { status: `failed:filesystem:md-write:${err.code || err.message}` };
  }
  try {
    _writeFileAtomic(sidecarPath, JSON.stringify(sidecar_json, null, 2) + "\n");
  } catch (err) {
    _safeUnlink(mdPath);
    return { status: `failed:filesystem:sidecar-write:${err.code || err.message}` };
  }
  return { status: "ok", mdPath, sidecarPath };
}

module.exports = { writeAtomicNote };
```

function _composeSidecar(input) {
  const generated_at = input.generated_at || new Date().toISOString();

  const surfaced_kinds = new Set();

    ...(input.ordered_blocks || []),
    ...(input.engagement_type_blocks || []),
  ];

```javascript
// platform/blueprints/cowork/helpers/kind-classifier-helper.js
//
// MCP auto-discovery: enumerate reachable namespaces + their tool lists,
// classify each into a known kind via deterministic pattern matching with
// optional LLM fallback for unrecognized namespaces. Cached per-namespace
// per-classifier-version. Returns { kind→namespaces, unclassified, new_since_last_fire }.

const fs = require("node:fs");
const path = require("node:path");

const _patternsCache = new Map();
function _loadPatterns(patternsPath) {
  if (_patternsCache.has(patternsPath)) return _patternsCache.get(patternsPath);
  const raw = fs.readFileSync(patternsPath, "utf8");
  const parsed = JSON.parse(raw);
  _patternsCache.set(patternsPath, parsed);
  return parsed;
}

function _globToRegex(glob) {
  const escaped = glob.replace(/[.+?^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*");
  return new RegExp(`^${escaped}$`, "i");
}

function _matchScore(kindConfig, namespace, toolNames) {
  let score = 0;
  for (const pat of kindConfig.namespace_patterns || []) {
    if (_globToRegex(pat).test(namespace)) score += 2;
  }
  for (const pat of kindConfig.tool_name_patterns || []) {
    const rx = _globToRegex(pat);
    for (const tool of toolNames) {
      if (rx.test(tool)) score += 1;
    }
  }
  return score;
}

function classifyConnectedKinds({
  reachable_namespaces,
  tools_by_namespace,
  vault_root,
  patterns_path,
  cache_path,
  llm_fallback,
} = {}) {
  const resolvedPatternsPath = patterns_path || path.join(vault_root, ".claude/skills/cowork/data/kind-patterns.json");
  const resolvedCachePath = cache_path || path.join(vault_root, ".claude/skills/cowork/data/kind-classifier-cache.json");
  const patterns = _loadPatterns(resolvedPatternsPath);
  const cache = fs.existsSync(resolvedCachePath)
    ? JSON.parse(fs.readFileSync(resolvedCachePath, "utf8"))
    : { schema_version: "1.0.0", entries: {}, classifier_version: "0.96.0" };

  const result = {
    kinds: {},
    unclassified: [],
    new_since_last_fire: [],
    classifier_cache_hit: true,
  };

  for (const ns of reachable_namespaces) {
    const toolNames = tools_by_namespace[ns] || [];
    const cached = cache.entries[ns];
    if (cached && cached.classifier_version === cache.classifier_version) {
      const kind = cached.kind;
      if (kind === "unclassified") {
        result.unclassified.push(ns);
      } else {
        (result.kinds[kind] ||= []).push(ns);
      }
      continue;
    }
    result.classifier_cache_hit = false;
    result.new_since_last_fire.push(ns);

    let bestKind = null;
    let bestScore = 0;
    for (const [kind, kindConfig] of Object.entries(patterns.kinds)) {
      const score = _matchScore(kindConfig, ns, toolNames);
      if (score > bestScore) {
        bestScore = score;
        bestKind = kind;
      }
    }

    let assignedKind;
    if (bestScore >= 2) {
      assignedKind = bestKind;
    } else if (typeof llm_fallback === "function") {
      assignedKind = llm_fallback({ namespace: ns, tool_names: toolNames }) || "unclassified";
    } else {
      assignedKind = "unclassified";
    }

    cache.entries[ns] = {
      kind: assignedKind,
      classified_at: new Date().toISOString(),
      classifier_version: cache.classifier_version,
      best_score: bestScore,
    };

    if (assignedKind === "unclassified") {
      result.unclassified.push(ns);
    } else {
      (result.kinds[assignedKind] ||= []).push(ns);
    }
  }

  fs.writeFileSync(resolvedCachePath, JSON.stringify(cache, null, 2) + "\n", "utf8");
  return result;
}

module.exports = { classifyConnectedKinds, _globToRegex, _matchScore };
```

// platform/blueprints/cowork/helpers/learn-from-checks-helper.js
//
// Rail L per-kind preference learning. Update formula per design §5.3.
// Pure data — no MCP calls, no LLM call.

const fs = require("node:fs");
const path = require("node:path");

const DEFAULTS = {
  decay: 0.98,
  lr: 0.15,
  smoothing: 5,
  clamp_min: 0.10,
  clamp_max: 3.00,
  warmup_days: 7,
  warmup_observations: 7,
  weight_deviation_threshold: 0.20,
  must_surface_backstop_days: 14,
};

const _round = (n, places = 3) => Math.round(n * 10 ** places) / 10 ** places;
const _clamp = (n, min, max) => Math.max(min, Math.min(max, n));

function updateWeights(prev_per_kind, observations, opts = {}) {
  const cfg = { ...DEFAULTS, ...opts };
  const next = { ...prev_per_kind };
  const counts = new Map();
  for (const obs of observations) {
    const c = counts.get(obs.kind) || { ticks: 0, skips: 0 };
    if (obs.ticked) c.ticks += 1;
    else c.skips += 1;
    counts.set(obs.kind, c);
  }
  for (const [kind, c] of counts) {
    const prev = next[kind] || { weight: 1.00, ticks: 0, skips: 0, warmup: true };
    const w_old = prev.weight;
    const w_raw = w_old * cfg.decay
      + cfg.lr * (c.ticks - 0.5 * c.skips) / (c.ticks + c.skips + cfg.smoothing);
    const w_new = _clamp(w_raw, cfg.clamp_min, cfg.clamp_max);
    next[kind] = {
      weight: _round(w_new, 3),
      ticks: prev.ticks + c.ticks,
      skips: prev.skips + c.skips,
      warmup: prev.warmup,
      last_updated: new Date().toISOString().slice(0, 10),
    };
  }
  return next;
}

function evaluateWarmup(per_kind_state, days_since_first, opts = {}) {
  const cfg = { ...DEFAULTS, ...opts };
  const out = { ...per_kind_state };
  for (const [kind, state] of Object.entries(out)) {
    if (!state.warmup) continue;
    const observations = state.ticks + state.skips;
    if (days_since_first >= cfg.warmup_days && observations >= cfg.warmup_observations) {
      out[kind] = { ...state, warmup: false };
    }
  }
  return out;
}

const RATING_SENTINEL_RX = /<!--\s*cowork:rating-block\s+schema=([\d.]+)\s+cadence=([\w-]+)\s+day=(\d{4}-\d{2}-\d{2})\s*-->/;
const CHECKBOX_LINE_RX = /^>\s*-\s*\[([ xX])\]\s+(.+?)\s*$/;

function parseRatingCallout(markdown) {
  if (!markdown || typeof markdown !== "string") return null;
  const sentinelMatch = markdown.match(RATING_SENTINEL_RX);
  if (!sentinelMatch) return null;
  const lines = markdown.split("\n");
  const ticks = [];
  let inCallout = false;
  for (const line of lines) {
    if (/^>\s*\[!todo\]\+\s+Was today useful/i.test(line)) {
      inCallout = true;
      continue;
    }
    if (inCallout) {
      const m = line.match(CHECKBOX_LINE_RX);
      if (m) {
        ticks.push({ ticked: m[1].toLowerCase() === "x", kind: m[2].trim().toLowerCase() });
      } else if (line.includes("<!-- cowork:rating-block")) {
        break;
      } else if (!line.startsWith(">") && line.trim().length > 0) {
        break;
      }
    }
  }
  return {
    schema_version: sentinelMatch[1],
    cadence: sentinelMatch[2],
    day: sentinelMatch[3],
    ticks,
  };
}

function scanAtomicNotes({ atomic_notes_dir, engagement_id }) {
  if (!fs.existsSync(atomic_notes_dir)) return { observations: [], notes_scanned: 0, notes_with_any_tick: 0 };
  const observations = [];
  let notes_scanned = 0;
  let notes_with_any_tick = 0;
  const files = fs.readdirSync(atomic_notes_dir).filter((f) => f.endsWith(".md") && !f.includes(".cowork.json"));
  for (const file of files) {
    const fullPath = path.join(atomic_notes_dir, file);
    const md = fs.readFileSync(fullPath, "utf8");
    const sidecarPath = fullPath.replace(/\.md$/, ".cowork.json");
    let sidecar = null;
    try { sidecar = JSON.parse(fs.readFileSync(sidecarPath, "utf8")); } catch (_) { /* pre-cycle note */ }
    if (sidecar && sidecar.engagement_id && sidecar.engagement_id !== engagement_id) continue;
    const parsed = parseRatingCallout(md);
    if (!parsed) continue;
    notes_scanned += 1;
    const anyTick = parsed.ticks.some((t) => t.ticked);
    if (anyTick) notes_with_any_tick += 1;
    for (const t of parsed.ticks) {
      observations.push({ kind: t.kind, ticked: t.ticked });
    }
  }
  return { observations, notes_scanned, notes_with_any_tick };
}

module.exports = {
  DEFAULTS,
  updateWeights,
  evaluateWarmup,
  parseRatingCallout,
  scanAtomicNotes,
  _clamp,
  _round,
};
