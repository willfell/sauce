# Migration Lifecycle Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the ~30 note-content migration/heal functions a per-vault version-gate (skip a heal once the vault is past its `introduced_in`; fail-safe run on unknown prior version) and extract them — with their shared helpers — out of the 14.6k-LOC `install.js` into one cohesive `platform/install-migrations.js` module driven by an ordered registry.

**Architecture:** A new `platform/install-migrations.js` exports `{ semverLt, MIGRATIONS, runMigrations }`. `MIGRATIONS` is an ordered array of `{ name, introduced_in, scope, run(ctx) }`; the heal bodies + their `_`-helpers move into this file verbatim. `install.js` captures `priorVersion` from the consumer's existing `platform-installed.json:workshop_version`, threads it into `installItem`, and replaces the scattered `await apply*()` heal block (lines ~1151–1171 + the `applyFinanceMigrations` aggregator at ~5669–5690) with a single `await runMigrations(ctx)`.

**Tech Stack:** Zero-dependency Node (CommonJS); the existing Templater-context installer; the seed-vault migration-regression harness as the behavior-preservation oracle.

**Design:** `Docs/plans/2026-06-27-migration-lifecycle-design.md`. **Refinement vs spec:** extract to ONE focused module (`platform/install-migrations.js`), not one-file-per-heal — the heals share `_`-helpers + a strict order, so co-locating them avoids cross-file require puzzles while still removing them from the monolith. **Do NOT** hand-bump versions (the bumper does it). No `Co-Authored-By: Claude` trailer. Stage explicit files, never `git add -A`. `git add -f` for anything under `scripts/`.

## The behavior-preservation contract (read first)

The ONLY intended behavior change is the gate: a migration whose `scope` matches the installing item is **skipped** when `priorVersion` is valid semver `>= introduced_in`. Everything else must be byte-identical. Two oracles enforce this:
1. **`run-seed-migrations` stays GREEN** with the seed's `installed.workshop_version` set to `"0.0.0"` (gate runs everything → identical to today).
2. **Golden-master:** a captured snapshot of the seed install's materialized note bodies; after each extraction step it must match byte-for-byte.

Move heal bodies **verbatim** — no logic edits during the move. The gate lives ONLY in `runMigrations`, never inside a heal body.

```js
'use strict';
// install-migrations.js — the note-content migration/heal family, extracted from
// install.js (cycle: migration-lifecycle). Each migration is gated per-vault: it
// runs only when the vault's prior installed workshop_version is unknown or below
// the version the heal first shipped in (introduced_in). Fail-safe: unknown prior
// version => run. Heal bodies + their _-helpers are moved here VERBATIM; the gate
// lives only in runMigrations, never in a heal body.

// Compare dotted numeric versions ("0.135.0", 4-segment "0.105.0.2" tolerated).
// Returns true iff a < b. Non-string / unparseable => treated as -Infinity seg/0.
function semverLt(a, b) {
  const pa = String(a).split('.').map(n => parseInt(n, 10) || 0);
  const pb = String(b).split('.').map(n => parseInt(n, 10) || 0);
  const len = Math.max(pa.length, pb.length);
  for (let i = 0; i < len; i++) {
    const x = pa[i] || 0, y = pb[i] || 0;
    if (x < y) return true;
    if (x > y) return false;
  }
  return false; // equal
}

// Ordered registry. Populated family-by-family in later tasks. ctx shape:
// { tp, manifest, variables, history, git, priorVersion }.
const MIGRATIONS = [];

async function runMigrations(ctx) {
  const itemName = ctx && ctx.manifest && ctx.manifest.name;
  for (const m of MIGRATIONS) {
    // scope filter — mirrors each heal's internal `if (manifest.name !== X) return`
    // (silent, no history) so scope-mismatch is byte-identical to today.
    if (m.scope !== 'global' && m.scope !== itemName) continue;
    // version gate — only for scope-matching, gated (introduced_in != null) heals.
    if (m.introduced_in != null && ctx.priorVersion && !semverLt(ctx.priorVersion, m.introduced_in)) {
      ctx.history.push({ step: 'migration', name: m.name, action: 'skipped_version_gated',
        prior_version: ctx.priorVersion, introduced_in: m.introduced_in,
        git_commit: ctx.git && ctx.git.commit, git_tag: ctx.git && ctx.git.tag });
      continue;
    }
    try {
      await m.run(ctx);
    } catch (e) {
      ctx.history.push({ step: 'migration', name: m.name, action: 'error',
        message: String((e && e.message) || e),
        git_commit: ctx.git && ctx.git.commit, git_tag: ctx.git && ctx.git.tag });
    }
  }
}

module.exports = { semverLt, MIGRATIONS, runMigrations };
```

```js
'use strict';
// Installs the seed vault into a tmpdir and writes a sorted "<relpath>\t<sha256>"
// manifest of every materialized markdown note. The migration-lifecycle extraction
// must keep this byte-identical (the seed's prior version is 0.0.0 so every gated
// migration runs). Run: node scripts/capture-seed-goldenmaster.js [--check]
const fs = require('fs'); const os = require('os'); const path = require('path');
const crypto = require('crypto'); const { execFileSync } = require('child_process');
const REPO = path.join(__dirname, '..');
const SEED = path.join(REPO, 'platform/test/seed-vault');
const OUT = path.join(REPO, 'platform/test/fixtures/seed-goldenmaster.txt');

function copyDir(src, dst) { fs.mkdirSync(dst, { recursive: true });
  for (const e of fs.readdirSync(src, { withFileTypes: true })) {
    const s = path.join(src, e.name), d = path.join(dst, e.name);
    if (e.isDirectory()) copyDir(s, d); else fs.copyFileSync(s, d);
  } }
function walk(root, base, acc) { for (const e of fs.readdirSync(root, { withFileTypes: true })) {
    const p = path.join(root, e.name);
    if (e.isDirectory()) walk(p, base, acc);
    else if (e.name.endsWith('.md')) {
      const rel = path.relative(base, p);
      if (rel.startsWith('spice/') || rel.startsWith('ranch/templates/')) {
        acc.push(rel + '\t' + crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex'));
      }
    } } }
const tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'seed-gm-'));
copyDir(SEED, tmp);
execFileSync('node', [path.join(REPO, 'platform/install.js'), '--vault', tmp, '--auto-approve'],
  { stdio: 'inherit', maxBuffer: 64 * 1024 * 1024 });
const lines = []; walk(tmp, tmp, lines); lines.sort();
const manifest = lines.join('\n') + '\n';
fs.rmSync(tmp, { recursive: true, force: true });
if (process.argv.includes('--check')) {
  const prev = fs.existsSync(OUT) ? fs.readFileSync(OUT, 'utf8') : '';
  if (prev !== manifest) {
    console.error('GOLDEN-MASTER MISMATCH — the install materialization changed. Diff the seed vs fixture.');
    const a = prev.split('\n'), b = manifest.split('\n'); const set = new Set(a);
    for (const l of b) if (!set.has(l)) console.error('  + ' + l);
    const setB = new Set(b); for (const l of a) if (!setB.has(l)) console.error('  - ' + l);
    process.exit(1);
  }
  console.log('golden-master: OK (' + manifest.split('\n').filter(Boolean).length + ' notes byte-identical)');
} else { fs.writeFileSync(OUT, manifest); console.log('wrote ' + OUT + ' (' + lines.length + ' notes)'); }
```
