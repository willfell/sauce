```javascript
/**
 * ProjectDocsSections — v1.16.0
 *
 * Renders the Docs Hub view as one labeled bucket per section. Reads
 * `sections[]` from the parent project's frontmatter (default
 * [Knowledge, Notes]); for each section, queries doc-notes whose
 * `section:` matches OR whose folder slug matches, then renders via
 * BeaconCards in row layout. Trails an "Unfiled" bucket for orphans.
 */
class ProjectDocsSections {
  async render(dv, opts = {}) {
    const projectSlug = this._extractProjectSlug(dv);
    if (!projectSlug) return;

    const project = this._loadProject(dv, projectSlug);
    const sections = this._resolveSections(project);

    const docsRoot = `spice/projects/${projectSlug}/docs`;
    const allDocs = dv.pages(`"${docsRoot}"`)
      .where(p => p.type === "doc-note");

    const filterContext = opts.filterContext || null;
    const renderedSlugs = new Set();

    for (const section of sections) {
      const slug = this._slugify(section);
      renderedSlugs.add(slug);
      const inSection = allDocs.where(p =>
        p.section === section ||
        (p.file.folder || "").endsWith(`/${slug}`)
      );
      await this._renderBucket(dv, section, slug, inSection, projectSlug, filterContext);
    }

    const orphans = allDocs.where(p => {
      const folderSlug = (p.file.folder || "").split("/").pop();
      return !renderedSlugs.has(folderSlug) && !renderedSlugs.has(this._slugify(p.section || ""));
    });
    if (orphans.length > 0) {
      await this._renderUnfiledBucket(dv, orphans, sections, projectSlug);
    }
  }

  _extractProjectSlug(dv) {
    const folder = dv.current()?.file?.folder || "";
    const match = folder.match(/^spice\/projects\/([^/]+)\/docs$/);
    return match ? match[1] : null;
  }

  _loadProject(dv, slug) {
    const candidates = dv.pages(`"spice/projects/${slug}"`)
      .where(p => p.type === "project");
    return candidates.length ? candidates[0] : null;
  }

  _resolveSections(project) {
    if (project && Array.isArray(project.sections) && project.sections.length > 0) {
      return project.sections.map(String);
    }
    return ["Knowledge", "Notes"];
  }

  _slugify(label) {
    return String(label || "")
      .toLowerCase()
      .trim()
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/^-+|-+$/g, "");
  }

  async _renderBucket(dv, label, slug, pages, projectSlug, filterContext) {
    dv.header(3, label);
    const newButtonContainer = dv.el("div", "");
    await customJS.EntityCreate.render(dv, {
      instance: "doc-note",
      presetPrompts: { section: label }
    });

    if (pages.length === 0) {
      dv.paragraph(`> [!example]+ No docs in **${label}** yet — use **+ New ${label}** above to create one.`);
      return;
    }

    customJS.BeaconCards.render(dv, {
      pages,
      layout: "row",
      icon: "file-text",
      meta: (p) => {
        const createdRaw = p.created_at || p.file.ctime;
        const created = window.moment(createdRaw).format("MMM D");
        const edited = window.moment(p.file.mtime?.ts || p.file.mtime).fromNow();
        return `created ${created} · edited ${edited}`;
      }
    });
  }

  async _renderUnfiledBucket(dv, orphans, declaredSections, projectSlug) {
    dv.header(3, "Unfiled");
    dv.paragraph(`> [!warning]+ ${orphans.length} doc(s) below have a section that isn't declared in this project's \`sections[]\`. Edit the project's frontmatter or move the file into a declared bucket to reconcile.`);
    customJS.BeaconCards.render(dv, {
      pages: orphans,
      layout: "row",
      icon: "alert-triangle",
      meta: (p) => `section: ${p.section || "(none)"} · folder: ${p.file.folder || "(root)"}`
    });
  }
}
```

```javascript
/**
 * ProjectMeetingsPanel — v1.16.0
 *
 * Renders the "## Meetings" section on a project note. Queries every
 * meeting in spice/meetings/notes whose `project:` frontmatter is a
 * wikilink to the current project. Shows the 5 most-recent (date desc)
 * with an inline "view all N" toggle if more exist. + New meeting for
 * this project button uses EntityCreate presetPrompts so the project
 * is pre-selected.
 */
class ProjectMeetingsPanel {
  async render(dv, opts = {}) {
    const currentPath = dv.current()?.file?.path;
    const projectName = dv.current()?.file?.name;
    if (!currentPath || !projectName) return;

    await customJS.EntityCreate.render(dv, {
      instance: "meeting",
      presetPrompts: { project: `[[${projectName}]]` }
    });

    const meetings = dv.pages('"spice/meetings/notes"')
      .where(p => p.type === "meeting" && this._projectMatches(p.project, currentPath, projectName));

    if (meetings.length === 0) {
      dv.paragraph(`> [!info]+ No meetings linked to this project yet. Use **+ New meeting for this project** above, or add \`project: "[[${projectName}]]"\` to an existing meeting's frontmatter.`);
      return;
    }

    const sorted = meetings.sort(p => p.date, "desc");
    const top5 = sorted.limit(5);
    const total = meetings.length;

    customJS.BeaconCards.render(dv, {
      pages: top5,
      layout: "row",
      icon: "calendar",
      meta: (p) => {
        const date = p.date ? String(p.date).slice(0, 10) : "(no date)";
        const attendees = Array.isArray(p.attendees) ? p.attendees.length : 0;
        return `${date} · ${attendees} attendee${attendees === 1 ? "" : "s"}`;
      }
    });

    if (total > 5) {
      const expandEl = dv.el("p", `View all ${total} meetings →`, { cls: "project-meetings-expand" });
      expandEl.style.cursor = "pointer";
      expandEl.addEventListener("click", () => {
        customJS.BeaconCards.render(dv, {
          pages: sorted,
          layout: "row",
          icon: "calendar",
          meta: (p) => {
            const date = p.date ? String(p.date).slice(0, 10) : "(no date)";
            const attendees = Array.isArray(p.attendees) ? p.attendees.length : 0;
            return `${date} · ${attendees} attendee${attendees === 1 ? "" : "s"}`;
          }
        });
        expandEl.remove();
      });
    }
  }

  _projectMatches(field, currentPath, projectName) {
    if (!field) return false;
    if (typeof field === "string") {
      return field.includes(`[[${projectName}]]`) || field === projectName;
    }
    if (field.path) return field.path === currentPath;
    if (field.display) return field.display === projectName;
    return false;
  }
}
```

```javascript
// applyProjectSectionsMigration — v0.102.0. Auto-migrate existing flat
// docs/*.md to docs/knowledge/<doc>.md, add section: "Knowledge" frontmatter,
// detect existing subfolders containing doc-notes and register them as
// custom sections in the parent project's frontmatter sections[]. Idempotent
// (skipped per-project if docs/knowledge/ already exists). Failure-loud
// per-project (catches + logs, never throws).
async function applyProjectSectionsMigration(tp, manifest, variables, history, git) {
  if (!manifest || manifest.name !== "project") return;
  if (!tp || !tp.app || !tp.app.vault || !tp.app.vault.adapter) return;
  const adapter = tp.app.vault.adapter;

  const projectsRoot = "spice/projects";
  if (!(await adapter.exists(projectsRoot))) return;

  let projectsList;
  try {
    projectsList = await adapter.list(projectsRoot);
  } catch (e) {
    history?.push({
      event: "warning",
      step: "project_sections_migration",
      name: "project",
      reason: `list failed for ${projectsRoot}: ${e.message}`,
      git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
      attempted_at: new Date().toISOString(),
    });
    return;
  }

  const projectDirs = (projectsList.folders || [])
    .filter(d => d.split("/").pop() !== "All Projects");

  let migratedCount = 0;
  let skippedCount = 0;
  let warnCount = 0;

  for (const projectDir of projectDirs) {
    const docsDir = `${projectDir}/docs`;
    const knowledgeDir = `${docsDir}/knowledge`;
    try {
      if (!(await adapter.exists(docsDir))) { skippedCount += 1; continue; }
      if (await adapter.exists(knowledgeDir)) { skippedCount += 1; continue; }  // idempotency guard

      // Create knowledge/ and notes/ folders
      await adapter.mkdir(knowledgeDir).catch(() => {});
      const notesDir = `${docsDir}/notes`;
      await adapter.mkdir(notesDir).catch(() => {});

      // List existing docs/ contents
      const docsList = await adapter.list(docsDir);
      const flatDocs = (docsList.files || []).filter(f => {
        const basename = f.split("/").pop();
        return basename !== "Docs.md" && basename.endsWith(".md");
      });

      // Move each flat doc into knowledge/, adding section: frontmatter
      for (const flatPath of flatDocs) {
        const basename = flatPath.split("/").pop();
        const newPath = `${knowledgeDir}/${basename}`;
        const body = await adapter.read(flatPath);
        const patched = _ensureSectionFrontmatter(body, "Knowledge");
        await adapter.write(newPath, patched);
        await adapter.remove(flatPath);
      }

      // Detect existing custom subfolders + collect for sections[]
      const customSections = [];
      for (const sub of (docsList.folders || [])) {
        const slug = sub.split("/").pop();
        if (slug === "knowledge" || slug === "notes") continue;
        // Only count subfolders containing at least one doc-note
        const subList = await adapter.list(sub);
        let hasDocNote = false;
        for (const fp of (subList.files || [])) {
          if (!fp.endsWith(".md")) continue;
          const fb = await adapter.read(fp);
          if (/^type:\s*["']?doc-note["']?\s*$/m.test(fb)) { hasDocNote = true; break; }
        }
        if (hasDocNote) {
          const label = _titlecaseFromSlug(slug);
          customSections.push(label);
          // Also add section: frontmatter to each doc in this subfolder (if absent)
          for (const fp of (subList.files || [])) {
            if (!fp.endsWith(".md")) continue;
            const body = await adapter.read(fp);
            if (!/^section:/m.test(body)) {
              await adapter.write(fp, _ensureSectionFrontmatter(body, label));
            }
          }
        }
      }

      // Update parent project's sections[] frontmatter (only if absent)
      const projectName = projectDir.split("/").pop();
      const projectNotePath = `${projectDir}/${_titlecaseFromSlug(projectName)}.md`;
      if (await adapter.exists(projectNotePath)) {
        const pBody = await adapter.read(projectNotePath);
        if (!/^sections:/m.test(pBody)) {
          const defaultSections = ["Knowledge", "Notes", ...customSections];
          const patched = _ensureSectionsFrontmatter(pBody, defaultSections);
          await adapter.write(projectNotePath, patched);
        }
      }

      migratedCount += 1;
      history?.push({
        event: "info",
        step: "project_sections_migration",
        name: "project",
        target: projectDir,
        action: "migrated",
        custom_sections: customSections,
        git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
        attempted_at: new Date().toISOString(),
      });
    } catch (e) {
      warnCount += 1;
      history?.push({
        event: "warning",
        step: "project_sections_migration",
        name: "project",
        reason: `migration failed for ${projectDir}: ${e?.message || String(e)}`,
        git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
        attempted_at: new Date().toISOString(),
      });
    }
  }

  history?.push({
    event: "info",
    step: "project_sections_migration",
    name: "project",
    reason: `migrated ${migratedCount} project(s); skipped ${skippedCount}; ${warnCount} warning(s)`,
    git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
    attempted_at: new Date().toISOString(),
  });
}

function _ensureSectionFrontmatter(body, sectionLabel) {
  if (/^section:/m.test(body)) return body;
  const fmMatch = body.match(/^---\n([\s\S]*?)\n---/);
  if (!fmMatch) return body;
  const fm = fmMatch[1];
  const newFm = `${fm}\nsection: "${sectionLabel}"`;
  return body.replace(/^---\n[\s\S]*?\n---/, `---\n${newFm}\n---`);
}

function _ensureSectionsFrontmatter(body, sections) {
  if (/^sections:/m.test(body)) return body;
  const fmMatch = body.match(/^---\n([\s\S]*?)\n---/);
  if (!fmMatch) return body;
  const fm = fmMatch[1];
  const yamlList = sections.map(s => `  - ${s}`).join("\n");
  const newFm = `${fm}\nsections:\n${yamlList}`;
  return body.replace(/^---\n[\s\S]*?\n---/, `---\n${newFm}\n---`);
}

function _titlecaseFromSlug(slug) {
  return String(slug)
    .split("-")
    .map(w => w.charAt(0).toUpperCase() + w.slice(1))
    .join(" ");
}
```

```javascript
// applyVaultDefaultPaths — v0.102.0. Ensures spice/resources/{notes,attachments}/
// exist; configures Obsidian's newFileLocation/newFileFolderPath/attachmentFolderPath
// in .obsidian/app.json IFF currently absent or set to vault-root defaults. Never
// clobbers user customizations.
async function applyVaultDefaultPaths(tp, history, git) {
  if (!tp || !tp.app || !tp.app.vault || !tp.app.vault.adapter) return;
  const adapter = tp.app.vault.adapter;

  // 1. Ensure folders exist
  for (const folder of ["spice/resources", "spice/resources/notes", "spice/resources/attachments"]) {
    try {
      if (!(await adapter.exists(folder))) {
        await adapter.mkdir(folder);
      }
      const gitkeep = `${folder}/.gitkeep`;
      if (!folder.endsWith("resources") && !(await adapter.exists(gitkeep))) {
        await adapter.write(gitkeep, "");
      }
    } catch (e) {
      history?.push({
        event: "warning", step: "vault_default_paths",
        reason: `mkdir failed for ${folder}: ${e.message}`,
        git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
        attempted_at: new Date().toISOString(),
      });
    }
  }

  // 2. Read-modify-write .obsidian/app.json
  const appJsonPath = ".obsidian/app.json";
  let app;
  try {
    if (!(await adapter.exists(appJsonPath))) {
      app = {};
    } else {
      const raw = await adapter.read(appJsonPath);
      app = raw ? JSON.parse(raw) : {};
    }
  } catch (e) {
    history?.push({
      event: "warning", step: "vault_default_paths",
      reason: `failed to read ${appJsonPath}: ${e.message}`,
      git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
      attempted_at: new Date().toISOString(),
    });
    return;
  }

  let changed = false;

  // newFileLocation: only write if absent or set to "root"
  if (!Object.prototype.hasOwnProperty.call(app, "newFileLocation") || app.newFileLocation === "root") {
    app.newFileLocation = "folder";
    changed = true;
  }

  // newFileFolderPath: only write if absent or empty string
  if (!app.newFileFolderPath) {
    app.newFileFolderPath = "spice/resources/notes";
    changed = true;
  }

  // attachmentFolderPath: only write if absent or empty string
  if (!app.attachmentFolderPath) {
    app.attachmentFolderPath = "spice/resources/attachments";
    changed = true;
  }

  if (changed) {
    try {
      await adapter.write(appJsonPath, JSON.stringify(app, null, 2));
      history?.push({
        event: "info", step: "vault_default_paths",
        action: "applied_canonical_defaults",
        git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
        attempted_at: new Date().toISOString(),
      });
    } catch (e) {
      history?.push({
        event: "warning", step: "vault_default_paths",
        reason: `write failed for ${appJsonPath}: ${e.message}`,
        git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
        attempted_at: new Date().toISOString(),
      });
    }
  } else {
    history?.push({
      event: "info", step: "vault_default_paths",
      reason: "all keys already customized; no changes",
      git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty,
      attempted_at: new Date().toISOString(),
    });
  }
}
```
