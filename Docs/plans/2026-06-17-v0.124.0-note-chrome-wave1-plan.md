# Note-chrome Wave 1 + Foundation Closure + Hygiene Implementation Plan (v0.124.0)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Broadcast the note-chrome standard to `meetings`, `scratch`, and daily `to-do` (breadcrumb trails + Meeting.md SectionLabel conformance), heal existing notes, close residual foundation items (lint gate, nav-buttons label ellipsis), and ship pre-adoption hygiene (project_name backfill, schemas-index parity) — all as one v0.124.0 release.

**Architecture:** Breadcrumbs are config-driven: each blueprint manifest declares a `breadcrumb: { types: {...} }` block that the installer's `applyBreadcrumb()` aggregates into `ranch/breadcrumb-registry.json` (no install.js change for new declarations). Chrome is live-rendered via `dv.view(customjs-guard, {class})`, so template edits affect NEW notes; existing notes are updated by an idempotent, `.sauce-backup`-snapshotting heal (`applyNoteChromeHeal`) modeled on `applyDocsHubButtonRepair`. A new `lint-note-chrome.js` gate enforces the grammar in `release:preflight`.

**Tech Stack:** Node.js (CommonJS) install/test scripts, JSON manifests, Obsidian Templater markdown templates, dataviewjs view dispatch, the seed-vault regression harness.

```javascript
// Idempotent body transform. Returns the new body, or the same string if no change.
function _healNoteChromeBody(body, type) {
  if (typeof body !== "string") return body;
  let out = body;
  // 1. Inject breadcrumb if absent. Insert before the first SpaceNavButtons block,
  //    OR (scratch-day) after the leading "# " H1. No "---" between breadcrumb and nav.
  if (!/class:\s*"Breadcrumb"/.test(out)) {
    const bc = '```dataviewjs\nawait dv.view("{{views_path}}/customjs-guard", { class: "Breadcrumb" });\n```\n\n';
    // NOTE: installed notes already have {{views_path}} substituted to ranch/views.
    const bcInstalled = bc.replace('{{views_path}}', 'ranch/views');
    const navIdx = out.indexOf('class: "SpaceNavButtons"');
    if (navIdx !== -1) {
      const fence = out.lastIndexOf('```dataviewjs', navIdx);
      if (fence !== -1) out = out.slice(0, fence) + bcInstalled + out.slice(fence);
    }
  }
  // 2. meeting only: rewrite "## Heading" content headers to SectionLabel.
  if (type === "meeting") {
    const map = [["Attendees", true], ["Agenda", false], ["Notes", false], ["Action Items", false]];
    for (const [text, top] of map) {
      const re = new RegExp("(?:^|\\n)(?:---\\s*\\n)?##\\s+" + text.replace(/ /g, "\\s+") + "\\s*\\n");
      if (re.test(out)) {
        const args = top ? `[{ text: "${text}", top: true }]` : `[{ text: "${text}" }]`;
        const sl = '\n```dataviewjs\nawait dv.view("ranch/views/customjs-guard", { class: "SectionLabel", args: ' + args + ' });\n```\n';
        out = out.replace(re, sl);
      }
    }
  }
  return out;
}
```

Then the driver (enumerate target notes by frontmatter `type`, snapshot to `.sauce-backup` before writing, per-note try/catch, history push, never throw):

```javascript
async function applyNoteChromeHeal(tp, history, git) {
  if (!tp || !tp.app || !tp.app.vault || !tp.app.vault.adapter) return;
  const adapter = tp.app.vault.adapter;
  const roots = ["spice/meetings", "spice/scratch", "spice/to-do"];
  let healed = 0, warned = 0;
  for (const root of roots) {
    if (!(await adapter.exists(root))) continue;
    const files = await _listMarkdownRecursive(adapter, root); // reuse/duplicate an existing recursive lister in install.js
    for (const fpath of files) {
      try {
        const before = await adapter.read(fpath);
        const type = _frontmatterType(before); // small helper: parse `type:` from FM
        if (!["meeting", "scratch", "scratch-day", "to-do"].includes(type)) continue;
        const after = _healNoteChromeBody(before, type);
        if (after === before) continue;
        await _snapshotToSauceBackup(adapter, fpath, before); // mirror applyFinanceMigrations snapshot
        await adapter.write(fpath, after);
        healed += 1;
        if (history) history.push({ event: "info", step: "note_chrome_heal", target: fpath, action: "healed",
          git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty, attempted_at: new Date().toISOString() });
      } catch (e) {
        warned += 1;
        if (history) history.push({ event: "warning", step: "note_chrome_heal", reason: `${fpath}: ${e && e.message ? e.message : String(e)}`,
          git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty, attempted_at: new Date().toISOString() });
      }
    }
  }
  if (history) history.push({ event: "info", step: "note_chrome_heal", reason: `healed ${healed}; ${warned} warning(s)`,
    git_commit: git.commit, git_tag: git.tag, git_dirty: git.dirty, attempted_at: new Date().toISOString() });
}
```
