- [ ] **S1.3** Implement `recurrence-parser.js`. Class declaration (CustomJS convention — single class on `customJS.RecurrenceParser`). API:

```js
class RecurrenceParser {
  // static API; instance is irrelevant
  static matches(grammar, dateMoment, registryCreatedAtMoment) { ... }
  static isSupported(grammar) { ... }  // used by dialog validation
}
```

  Parsing strategy: lowercase the grammar string; strip the leading `every `; switch on first token:
  - `day` → always true.
  - Day-name set (`monday`, `mon`, `tuesday`, `tue`, etc., possibly comma- or space-separated) → set membership.
  - `weekday` → 1..5 in moment's `day()` (Mon=1, Fri=5).
  - `weekend` → 0 or 6.
  - `<n>(st|nd|rd|th) of month` → `dateMoment.date() === n`.
  - `<n> weeks on <weekday>` → match weekday AND `dateMoment.diff(registryCreatedAtMoment, 'days') % (n * 7) === 0`.
  - Any other → return `false`.

  Day-name normalization: accept full names (`Monday`) and 3-letter shorts (`Mon`); case-insensitive.

  `isSupported(grammar)` reuses the same parse and returns true if the grammar parses to a known family, even without evaluating against a date. Used by the dialog.

```js
class ToDoDailyCarryover {
  async render(dv) {
    if (dv.container.closest('.markdown-embed')) return;
    const ren = (window.__toDoCarryoverGen = (window.__toDoCarryoverGen || 0) + 1);
    const cur = dv.current();
    if (!cur || !cur.file || cur.type !== 'to-do') return;
    const filename = cur.file.name;  // e.g. ToDo-2026-06-15
    const todayStr = filename.replace(/^ToDo-/, '');
    if (!/^\d{4}-\d{2}-\d{2}$/.test(todayStr)) return;
    await this.materialize(cur.file.path, todayStr);
    // render nothing visible by default; opt-in audit chip via opts later
  }

  static findPreviousDaily(todayStr, vault) {
    const moment = window.moment;
    const today = moment(todayStr, 'YYYY-MM-DD');
    for (let n = 1; n <= 7; n++) {
      const candidate = today.clone().subtract(n, 'days');
      const dStr = candidate.format('YYYY-MM-DD');
      const dPath = `spice/to-do/${candidate.format('YYYY/MM-MMMM')}/ToDo-${dStr}.md`;
      const af = vault.getAbstractFileByPath ? vault.getAbstractFileByPath(dPath) : null;
      if (af) return { path: dPath, dateStr: dStr };
    }
    return null;
  }

  static extractMigratableBlocks(content) {
    const all = customJS.TaskParser.parseTasks(content);
    return all.filter(b => !/\[project::\s*\[\[/.test(b.topLine));
  }

  static applyToToday(todayContent, migratedBlocks, sourceDate) {
    // 1. Build the `## Carryover (from YYYY-MM-DD)` heading + decorated blocks.
    // 2. Find the ToDoDailyCarryover dataviewjs block in todayContent.
    // 3. Insert the heading + lines after that block; before the next dataviewjs block (or EOF).
    // 4. Decorate each top-line: ensure `[from:: [[ToDo-<sourceDate>]]]` appended.
    // 5. Write sentinel `<!-- carryover-from-<sourceDate> -->` into frontmatter region.
    // 6. Return new content.
  }

  async materialize(todayPath, todayStr) {
    const vault = app.vault;
    const todayFile = vault.getAbstractFileByPath(todayPath);
    if (!todayFile) return;
    const todayContent = await vault.read(todayFile);
    // Idempotency guard via sentinel substring check.
    if (todayContent.includes('<!-- carryover-from-')) return;
    const prior = ToDoDailyCarryover.findPreviousDaily(todayStr, vault);
    if (!prior) {
      // No prior within window — write the (none) sentinel + no-op.
      const updated = ToDoDailyCarryover._writeSentinel(todayContent, '(none)');
      await vault.modify(todayFile, updated);
      return;
    }
    const priorFile = vault.getAbstractFileByPath(prior.path);
    const priorContent = await vault.read(priorFile);
    const blocks = ToDoDailyCarryover.extractMigratableBlocks(priorContent);
    if (!blocks.length) {
      const updated = ToDoDailyCarryover._writeSentinel(todayContent, prior.dateStr);
      await vault.modify(todayFile, updated);
      return;
    }
    // Atomic-pair write: prior strip + today insert.
    const result = customJS.TaskParser.applyMigration(priorContent, todayContent, blocks.map(b => b.index), {
      lineDecorator: (line) => `${line.trim()} [from:: [[ToDo-${prior.dateStr}]]]`,
      destSectionHeading: `## Carryover (from ${prior.dateStr})`,
      destAnchor: { afterClass: 'ToDoDailyCarryover' }
    });
    await vault.modify(priorFile, result.source);
    const withSentinel = ToDoDailyCarryover._writeSentinel(result.dest, prior.dateStr);
    await vault.modify(todayFile, withSentinel);
  }

  static _writeSentinel(content, sourceDate) {
    // Insert `<!-- carryover-from-<sourceDate> -->` after the second `---\n`.
  }
}
```

```js
class ToDoDailyProjectGroups {
  async render(dv) {
    if (dv.container.closest('.markdown-embed')) return;
    const cur = dv.current();
    if (!cur || cur.type !== 'to-do') return;
    const projects = dv.pages('"spice/projects"').where(p => p.type === 'project').array();
    const blocks = [];
    for (const proj of projects) {
      const projSlug = proj.project_slug || proj.file.name.toLowerCase().replace(/[^a-z0-9]+/g, '-');
      const projName = proj.file.name;
      const tasks = [];
      // (a) Project-todo note's Owned Tasks.
      const todoPath = `spice/projects/${projSlug}/${projName} To-Do.md`;
      const todoPage = dv.page(todoPath);
      if (todoPage && todoPage.file && todoPage.file.tasks) {
        for (const t of todoPage.file.tasks) {
          if (!t.completed) tasks.push({ task: t, source: todoPath });
        }
      }
      // (b) Meetings linked to this project — open tasks in meeting body.
      const meetings = dv.pages('"spice/meetings/notes"').where(m =>
        m.project && String(m.project).replace(/^\[\[|\]\]$/g, '') === projName
      ).array();
      for (const meet of meetings) {
        if (meet.file && meet.file.tasks) {
          for (const t of meet.file.tasks) {
            if (!t.completed) tasks.push({ task: t, source: meet.file.path });
          }
        }
      }
      if (tasks.length === 0) continue;
      blocks.push({ projSlug, projName, projPath: proj.file.path, tasks });
    }
    if (blocks.length === 0) return;  // empty section renders nothing
    // Render ## Open Project Tasks + per-project sub-blocks
    const sectionEl = dv.container.createEl('h2', { text: 'Open Project Tasks' });
    for (const blk of blocks) {
      // Anchor comment for ToDoDailyRecurring's project routing:
      dv.container.appendChild(document.createComment(` project-group-anchor-${blk.projSlug} `));
      const h3 = dv.container.createEl('h3');
      const a = h3.createEl('a', { text: blk.projName });
      a.classList.add('internal-link');
      a.setAttribute('data-href', blk.projPath);
      a.onclick = (ev) => { ev.preventDefault(); app.workspace.openLinkText(blk.projPath, '', false); };
      const ul = dv.container.createEl('ul');

        li.style.cursor = 'pointer';
        li.onclick = () => app.workspace.openLinkText(t.source, '', false);
      }
    }
```

```js
async function applyToDoBlueprintMigration(tp, manifest, variables, history, git) {
  const adapter = tp.app.vault.adapter;
  const toDoRoot = 'spice/to-do';
  let migrated = 0, alreadyCurrent = 0, absorbedTasksHeading = 0;
  // Walk YYYY/MM-MMMM/ToDo-*.md recursively under toDoRoot.
  const dailies = await _walkToDoFiles(adapter, toDoRoot);
  for (const relPath of dailies) {
    try {
      const body = await adapter.read(relPath);
      // Skip if already v0.4.0 (marker substring).
      if (body.includes('class: "ToDoDailyCarryover"')) { alreadyCurrent++; continue; }
      // Detect v0.3.3 shape — 2 dataviewjs blocks, free-form below.
      if (!_isV033Shape(body)) { alreadyCurrent++; continue; }
      // Backup-before-write.
      await _saveSauceBackup(adapter, relPath, body);
      // Reshape:
      // 1. Parse: frontmatter + 2 dataviewjs blocks + body.
      // 2. Build new body: same frontmatter + nav block + leaf-actions block + `## Today's Capture\n` + free-form lines + 4 new dataviewjs blocks.
      // 3. If `## Tasks` heading present, drop the heading and absorb lines as Capture (absorbedTasksHeading++).
      // 4. If `## Notes` heading present, preserve verbatim AFTER all the dataviewjs blocks.
      const newBody = _reshapeToV040(body);
      await adapter.write(relPath, newBody);
      migrated++;
    } catch (e) {
      history.push({ ts: Date.now(), level: 'error', step: 'to_do_blueprint_migration', file: relPath, message: String(e.message || e) });
    }
  }
  history.push({ ts: Date.now(), level: 'info', step: 'to_do_blueprint_migration', migrated_count: migrated, already_current_count: alreadyCurrent, absorbed_tasks_heading_count: absorbedTasksHeading });
}
```

```js
async function applyProjectTodoBackfill(tp, manifest, variables, history, git) {
  const adapter = tp.app.vault.adapter;
  const projRoot = 'spice/projects';
  let created = 0, skipped = 0;
  // List project subfolders.
  const projDirs = await _listSubdirs(adapter, projRoot);
  for (const projSlug of projDirs) {
    // Discover the project's hub note (a `*.md` with `type: project` frontmatter).
    const hubName = await _findProjectHubName(adapter, projRoot + '/' + projSlug);
    if (!hubName) continue;
    const toDoPath = `${projRoot}/${projSlug}/${hubName} To-Do.md`;
    if (await adapter.exists(toDoPath)) { skipped++; continue; }
    // Render template inline (no Templater since we're in install context).
    const body = _renderProjectTodoTemplate({
      project_link: `[[${hubName}]]`,
      slug: projSlug,
      vault_identity_tag: variables.vault_identity_tag || '',
      views_path: variables.views_path,
    });
    await adapter.write(toDoPath, body);
    created++;
  }
  history.push({ ts: Date.now(), level: 'info', step: 'project_todo_backfill', created_count: created, skipped_count: skipped });
}
```

- **Consumer install RED at S20:** check landmine #16 — per-item bump rule. Confirm to-do blueprint version is 0.4.0 in BOTH platform/manifest.json blueprints[] AND platform/blueprints/to-do/manifest.json AND consumer's ranch/platform-subscription.json. All three must agree.
