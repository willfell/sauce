```js
      return { status: `failed:input:unknown-callout-type:${b.callout_type}:${i}` };
    }
  }
  for (let i = 0; i < input.engagement_type_blocks.length; i++) {

    if (!b || typeof b !== "object") return { status: `failed:input:malformed-engagement-type-block:${i}:not-object` };
```

```js
function _wrapCallout({ callout_type, title, body_md }) {
  // Strip trailing whitespace from each line; preserve internal blanks
  const bodyLines = body_md.split("\n").map(l => l.replace(/[ \t]+$/, ""));
  // Prefix each line with "> "; blank lines become "> " (no trailing space after >)
  const prefixed = bodyLines.map(l => l === "" ? ">" : `> ${l}`);
  const header = `> [!${callout_type}]+ ${title}`;
  return [header, ...prefixed].join("\n");
}
```

```js
function _composeMemoryCluster(memory_callouts) {
  // Returns the top-of-body memory cluster (yesterday + overnight + echoes).
  // Backlink is rendered separately at end of body, not here.
  const mc = memory_callouts || {};
  const parts = [];
  if (typeof mc.yesterday_md === "string" && mc.yesterday_md.trim()) parts.push(mc.yesterday_md.trimEnd());
  if (typeof mc.overnight_md === "string" && mc.overnight_md.trim()) parts.push(mc.overnight_md.trimEnd());
  if (typeof mc.echoes_md === "string" && mc.echoes_md.trim()) parts.push(mc.echoes_md.trimEnd());
```

  assertions.push(synopsisFirst);

  // 3. Memory callouts — only when non-empty input
  const mc = input.memory_callouts || {};
  for (const f of ["yesterday_md", "overnight_md", "echoes_md"]) {
    if (typeof mc[f] === "string" && mc[f].trim()) {
      const firstLine = mc[f].split("\n")[0].replace(/[ \t]+$/, "");
      assertions.push(firstLine);
    }

  // 4. Per ordered_blocks[i]
  for (const b of input.ordered_blocks) {
    assertions.push(`> [!${b.callout_type}]+ ${b.title}`);
  }

  // 5. Per engagement_type_blocks[i]
